#!/usr/bin/env python3
"""Bind edition 3.4 book metadata to frozen, actually accepted local inputs.

This helper never builds Lean, executes book programs, typesets, renders, or
performs a remote operation. It is read-only unless --apply is supplied.
The context pins final canonical hashes and a stable receipt timestamp.
Old book evidence and the three canonical documents are immutable inputs.
"""

import argparse
from collections import Counter, defaultdict
from copy import deepcopy
from hashlib import sha256
import json
import os

BOOK = Path('physics/qm/book')
NEW = BOOK / 'evidence/v3.4'
KERNEL = Path('physics/qm/kernel/evidence/v31.8')
FOUNDATION = {'propext', 'Classical.choice', 'Quot.sound'}
BASELINE = '7d05d0369b6793e28f0e39efaa77c0c9887479d1'
SOURCE_SET = 'a34337cd75cd70b0f2e1125899c4dcd09aa7b132642adaad2c17ef8a69f12f6e'
LOCAL = 'LOCAL_VERIFIED_REMOTE_PENDING'
BODY_RULE = ('stable chapter/appendix anchor inclusive to next chapter/appendix '
             'anchor exclusive; final appendix through EOF; exact UTF-8 bytes')


def require(test, message):
    if not test:
        raise ValueError(message)


def digest(data):
    return sha256(data).hexdigest()


def json_bytes(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()


def identity(path, data):
    return {'path': str(path), 'sha256': digest(data), 'bytes': len(data)}


def parse_claims(text):
    lines = text.split('<a id="appendix-a"></a>', 1)[0].splitlines()
    chapter = None
    rows = []
    for i, line in enumerate(lines):
        heading = re.fullmatch(r'# (\d+)\. .+', line)
        if heading:
            chapter = int(heading.group(1))
        anchor = re.fullmatch(r'<a id="([a-z0-9-]+)"></a>', line)
        if not anchor or anchor.group(1).startswith('ch-'):
            continue
        result_id = anchor.group(1).upper()
        following = next((x for x in lines[i + 1:] if x.strip()), '')
        title = re.match(r'\*\*(.+?)\*\*', following)
        require(title and re.match(re.escape(result_id) + r'(?=$|[^A-Za-z0-9_-])',
                                   title.group(1)), 'Invalid numbered anchor: ' + result_id)
        rows.append({'id': result_id, 'chapter': chapter, 'title': title.group(1)})
    return rows


def body_slices(text):
    marks = list(re.finditer(r'^<a id="(ch-\d+|appendix-[a-f])"></a>$', text, re.M))
    expected = ['ch-' + str(n) for n in range(63)] + ['appendix-' + x for x in 'abcdef']
    require([m.group(1) for m in marks] == expected, 'Chapter/appendix anchor order differs')
    return {m.group(1): text[m.start():marks[i + 1].start() if i + 1 < len(marks) else len(text)]
            for i, m in enumerate(marks)}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root', type=Path, required=True)
    parser.add_argument('--context', type=Path, required=True)
    parser.add_argument('--apply', action='store_true')
    args = parser.parse_args()
    root = args.root.resolve()
    context_path = args.context.resolve()
    inputs = {}
    outputs = {}

    def read_abs(path):
        path = Path(path).resolve()
        data = path.read_bytes()
        if path in inputs:
            require(inputs[path] == data, 'Input changed during assembly: ' + str(path))
        inputs[path] = data
        return data

    context_raw = read_abs(context_path)
    context = json.loads(context_raw)
    require(context.get('edition') == '3.4', 'Context must explicitly pin edition 3.4')
    when = context['recorded_at_utc']
    require(isinstance(when, str) and re.fullmatch(r'2026-\d\d-\d\dT[^\n]+(?:Z|\+00:00)', when),
            'An actual stable UTC receipt timestamp is required')
    require(context.get('repository', 'MurphyHoops/UEOT') == 'MurphyHoops/UEOT', 'Repository differs')
    prior_root = Path(context.get('prior_root', str(root.parent / 'qm_v317_work'))).resolve()
    require(prior_root != root, 'Prior edition must be a separate read-only reference')

    def relpath(path):
        path = Path(path)
        require(not path.is_absolute() and '..' not in path.parts, 'Unsafe repository path: ' + str(path))
        require((root / path).resolve().is_relative_to(root), 'Path leaves repository: ' + str(path))
        return path

    def read(path):
        return read_abs(root / relpath(path))

    def obj(path):
        return json.loads(read(path))

    def rid(path):
        return identity(relpath(path).as_posix(), read(path))

    def planned(path, data):
        path = relpath(path)
        require(path not in outputs, 'Duplicate output: ' + str(path))
        require(path in {BOOK / 'README.md', BOOK / 'SOURCE_MANIFEST.json', BOOK / 'CHAPTER_DOSSIER.json'}
                or path.is_relative_to(NEW), 'Output outside A ownership: ' + str(path))
        require(path not in {BOOK / 'UEOT_PHYSICS.md', BOOK / 'UEOT_PHYSICS.tex', BOOK / 'UEOT_PHYSICS.pdf'},
                'Canonical documents are immutable inputs')
        outputs[path] = data
        return identity(path.as_posix(), data)

    def planned_json(path, data):
        return planned(path, json_bytes(data))

    def external_binding(name):
        archived = root / NEW / 'chapter-inputs' / f'qm318_chapter{name}_bindings.json'
        value = context.get('chapter_bindings', {}).get(str(name),
                    str(archived) if archived.is_file() else f'qm318_chapter{name}_bindings.json')
        path = Path(value)
        return path if path.is_absolute() else context_path.parent / path

    # All old evidence is checked byte-for-byte, then retained without a write.
    prior_metadata = {}
    snapshots = {}
    for name, snapshot in [('SOURCE_MANIFEST.json', 'source-manifest-prior-v3.3.json'),
                           ('CHAPTER_DOSSIER.json', 'dossier-prior-v3.3.json'),
                           ('README.md', 'readme-prior-v3.3.md')]:
        prior_raw = read_abs(prior_root / BOOK / name)
        require(read(BOOK / name) == prior_raw, 'Current metadata no longer matches the v3.3 baseline: ' + name)
        prior_metadata[name] = prior_raw
        snapshots[name] = planned(NEW / snapshot, prior_raw)
    old_manifest = json.loads(prior_metadata['SOURCE_MANIFEST.json'])
    old_dossier = json.loads(prior_metadata['CHAPTER_DOSSIER.json'])
    require(old_dossier['edition'] == '3.3' and old_dossier['current_release_stage'] == 'MAIN_VERIFIED',
            'Prior dossier is not the completed 3.3 edition')
    require(len(old_manifest['entries']) == 11, 'Expected all 11 original source entries')
    old_results = [x for ch in old_dossier['chapters'] for x in ch.get('numbered_results', [])]
    require(len(old_results) == 131 and len({x['id'] for x in old_results}) == 131,
            'Prior 131 numbered objects are incomplete')
    old_evidence = []
    for path in sorted((prior_root / BOOK / 'evidence').rglob('*')):
        if path.is_file():
            relative = path.relative_to(prior_root)
            original = read_abs(path)
            require(read(relative) == original, 'Historical evidence changed: ' + str(relative))
            old_evidence.append(identity(relative.as_posix(), original))
    require(old_evidence, 'No historical book evidence found')

    proof = obj(KERNEL / 'lean-verification.json')
    manifest = obj(KERNEL / 'source-manifest.json')
    inventory = obj(KERNEL / 'declaration-inventory.json')
    axioms = obj(KERNEL / 'proof-axioms.json')
    witnesses = obj(KERNEL / 'finite-witnesses/verification.json')
    require(proof['status'] == axioms['status'] == witnesses['status'] == 'PASS', 'Actual proof/audit gates not PASS')
    require(proof['source_count'] == manifest['source_count'] == len(manifest['sources']) == 89,
            'Actual source count differs from the accepted set')
    require(proof['source_set_sha256'] == manifest['source_set_sha256'] == SOURCE_SET,
            'Current source set differs from the accepted freeze')
    calculated = digest(json.dumps(manifest['sources'], sort_keys=True, separators=(',', ':')).encode())
    require(calculated == SOURCE_SET, 'Source-set aggregate is not derived from manifest rows')
    source_rows = {x['path']: x for x in manifest['sources']}
    require(len(source_rows) == 89, 'Duplicate source-manifest path')
    # The real build remains bound to its original inputs. A separately checked,
    # single documentation-word correction binds final raw publication bytes.
    publication_manifest_path = KERNEL / 'publication-source-manifest.json'
    bridge_path = KERNEL / 'publication-source-reconciliation.json'
    publication_manifest = obj(publication_manifest_path)
    bridge = obj(bridge_path)
    bridge_id = rid(bridge_path)
    require(bridge['status'] == 'PASS_COMMENT_ONLY' and bridge['additional_compiler_invocations'] == 0
            and bridge['scientific_claims_changed'] is False and bridge['core_changed'] is False,
            'Publication bridge does not have the authorized comment-only scope')
    unified_identity = deepcopy(bridge['unified_source_identity'])
    publication_identity = deepcopy(bridge['publication_source_identity'])
    publication_set = publication_identity['source_set_sha256']
    require(unified_identity['manifest'] == rid(KERNEL / 'source-manifest.json')
            and unified_identity['verification'] == rid(KERNEL / 'lean-verification.json')
            and unified_identity['source_count'] == 89 and unified_identity['source_set_sha256'] == SOURCE_SET
            and unified_identity['actual_build_jobs'] == proof['actual_build_jobs'], 'Bridge does not bind immutable actual unified inputs')
    require(publication_identity['manifest'] == rid(publication_manifest_path)
            and publication_identity['source_count'] == publication_manifest['source_count'] == 89
            and publication_set == publication_manifest['source_set_sha256'] == context['publication_source_set_sha256'],
            'Publication manifest/bridge/caller pin differs')
    require(digest(json.dumps(publication_manifest['sources'], sort_keys=True, separators=(',', ':')).encode()) == publication_set,
            'Publication source aggregate is not derived from final rows')
    published_rows = {x['path']: x for x in publication_manifest['sources']}
    require(len(published_rows) == 89 and published_rows.keys() == source_rows.keys(), 'Publication source paths differ')
    changed = sorted(p for p in source_rows if published_rows[p] != source_rows[p])
    only_changed = 'formalization/ueot-qm/UEOTQM/PurificationUniqueness.lean'
    require(changed == bridge['changed_paths'] == [only_changed]
            and bridge['unchanged_source_paths'] == 88 and len(bridge['changes']) == 1,
            'Only the authorized single-file documentation correction may differ')
    for path, row in published_rows.items():
        require(rid(path) == row, 'Final publication source bytes changed: ' + path)
    change = bridge['changes'][0]
    require(change['path'] == only_changed and change['before'] == source_rows[only_changed]
            and change['after'] == published_rows[only_changed], 'Bridge changed source identities differ')
    for key in ['before_snapshot', 'after_snapshot', 'exact_diff']:
        require(rid(change[key]['path']) == change[key], 'Bridge snapshot/diff identity changed')
    before_comment = read(change['before_snapshot']['path'])
    after_comment = read(change['after_snapshot']['path'])
    old_word = b'/-- Different finite auxiliary carriers admit actual isometric extensions into'
    new_word = b'/-- Different finite auxiliary carriers have actual isometric extensions into'
    require(digest(before_comment) == source_rows[only_changed]['sha256']
            and digest(after_comment) == published_rows[only_changed]['sha256']
            and before_comment.count(old_word) == 1
            and before_comment.replace(old_word, new_word) == after_comment == read(only_changed),
            'The exact whole-file change is not the approved one-word documentation replacement')
    require(change['erased_source_equal'] is True and change['declarations_diagnostics_and_instances_equal'] is True
            and bridge['all_final_source_bytes_verified'] is True
            and bridge['unified_receipts_preserved_original_bytes'] is True
            and bridge['actual_workflow_raw_proof_escape_regex_passed'] is True,
            'Publication bridge checks are incomplete')
    publication_identity.update({'status': 'LOCAL_COMMENT_ONLY_BRIDGE_REMOTE_PENDING', 'reconciliation': bridge_id,
                                 'additional_local_compiler_invocations': 0,
                                 'scope': 'Final raw bytes differ in one documentation word. The original 8799-job build remains bound exclusively to unified_source_identity; final raw CI is still pending.'})

    def bound_source_identity(path):
        return {**published_rows[path], 'unified_input_sha256': source_rows[path]['sha256'],
                'unified_input_bytes': source_rows[path]['bytes'],
                'publication_identity_relation': 'DOCUMENTATION_WORD_ONLY' if path in changed else 'BYTE_IDENTICAL',
                **({'publication_reconciliation': bridge_id} if path in changed else {})}
    require(proof['actual_build_exit_code'] == proof['actual_axiom_audit_exit_code'] == 0,
            'Build or audit exit code is not zero')
    require(proof['core_source_modified'] is False and proof['proof_escape_gate'] == 'PASS',
            'Core or proof-escape gate failed')
    require(proof['generic_modules'] == inventory['module_count'] == 35
            and proof['explicit_theorems'] == inventory['explicit_theorems'] == 747,
            'Verified generic declaration counts disagree')
    generic_names = set(axioms['declarations'])
    require(len(generic_names) == proof['actual_named_axiom_reports'] == inventory['audit_target_count'] == 985,
            'Actual generic named audit count disagrees')
    require(set(inventory['audit_targets']) == generic_names and len(inventory['audit_targets']) == 985,
            'Actual generic audit differs from complete unique inventory')
    require(all(set(x) <= FOUNDATION for x in axioms['declarations'].values()), 'Forbidden foundation in actual generic audit')
    declaration_source = {}
    for module, declarations in inventory['modules'].items():
        path = f'formalization/ueot-qm/UEOTQM/{module}.lean'
        require(source_rows[path]['sha256'] == inventory['module_identities'][path]['sha256'],
                'Module inventory is stale: ' + module)
        for declaration in declarations:
            require(declaration['name'] not in declaration_source, 'Duplicate explicit public declaration')
            declaration_source[declaration['name']] = path
    witness_names = set()
    for check in witnesses['checks']:
        require(check['status'] == check['audit']['status'] == 'PASS' and check['exit_code'] == 0,
                'A witness execution/audit did not PASS')
        require(check['audit']['complete_unique'] is True, 'Witness audit incomplete or non-unique')
        names = check['audit']['actual']
        expected = [x['name'] for x in witnesses['declarations'][check['name']]]
        require(list(names) == expected == check['audit']['expected'], 'Witness source-order audit differs')
        require(all(set(x) <= FOUNDATION for x in names.values()), 'Forbidden witness foundation')
        path = f"formalization/ueot-qm/scripts/witnesses/{check['name']}.lean"
        require(source_rows[path]['sha256'] == check['source']['sha256'], 'Witness source no longer matches execution')
        for name in names:
            require(name not in witness_names and name not in generic_names, 'Duplicate witness identity')
            witness_names.add(name)
            declaration_source[name] = path
    require(len(witness_names) == witnesses['witness_audit_count'] == 269 and len(witnesses['checks']) == 10,
            'Expected ten actual witness groups and 269 exact named audits')
    require(proof['runtime_test_methods'] == 100 and proof['all_four_demos_passed'] is True,
            'Actual runtime/demo gates differ')
    require(proof['instance_verification']['status'] == 'PASS', 'Actual certificate gate failed')

    # The caller must pin all three final canonical identities, independently of build receipts.
    canonical = {key: rid(BOOK / name) for key, name in
                 [('canonical', 'UEOT_PHYSICS.md'), ('pdf', 'UEOT_PHYSICS.pdf'), ('latex', 'UEOT_PHYSICS.tex')]}
    pins = context['canonical_hashes']
    for key, pin in [('canonical', 'markdown_sha256'), ('pdf', 'pdf_sha256'), ('latex', 'tex_sha256')]:
        require(canonical[key]['sha256'] == pins[pin], 'Canonical identity differs from final caller pin: ' + key)
    text = read(BOOK / 'UEOT_PHYSICS.md').decode()
    prior_text = read_abs(prior_root / BOOK / 'UEOT_PHYSICS.md').decode()
    require(digest(prior_text.encode()) == old_dossier['canonical']['sha256'], 'Prior canonical source changed')
    claims = parse_claims(text)
    require(claims[:131] == parse_claims(prior_text), 'Inherited actual numbered statements changed')
    require(len(claims) == 147, 'Expected 131 inherited and 16 new numbered results')
    require('version: "3.4"' in text, 'Canonical edition marker is not 3.4')
    slices = body_slices(text)
    by_id = {x['id']: x for x in claims}
    require(len(by_id) == 147, 'Duplicate numbered result identity')

    receipt_paths = {name: relpath(context.get('receipts', {}).get(name, str(NEW / filename)))
                     for name, filename in [('structure', 'book-structure.json'), ('pdf_build', 'pdf-build.json'),
                                            ('visual', 'pdf-visual-review.json'), ('render', 'pdf-render-manifest.json'),
                                            ('integration', 'canonical-integration.json'),
                                            ('canonical_adjustments', 'canonical-layout-adjustments.json'),
                                            ('collaborator', 'pdf-collaborator-review.json')]}
    receipts = {name: obj(path) for name, path in receipt_paths.items()}
    receipt_ids = {name: rid(path) for name, path in receipt_paths.items()}
    structure, build, visual, render, integration = (receipts[x] for x in ['structure', 'pdf_build', 'visual', 'render', 'integration'])
    md_hash = canonical['canonical']['sha256']
    pdf_hash = canonical['pdf']['sha256']
    require(structure['status'] == 'PASS' and structure['source_sha256'] == md_hash
            and structure['claims'] == claims, 'Actual structure receipt does not bind current numbered statements')
    require(structure['chapters'] == 63 and structure['appendices'] == 6
            and structure['numbered_results'] == 147 and structure['internal_links_valid'] is True
            and structure['registry_rows_bound_to_actual_anchors'] is True, 'Book structure gate failed')
    require(build['source_sha256'] == md_hash and build['pdf_sha256'] == pdf_hash
            and build['tex_sha256'] == canonical['latex']['sha256'], 'PDF build does not bind all final canonical bytes')
    require(len(build['compile_passes']) == 3 and all(x['exit_code'] == 0 for x in build['compile_passes'])
            and build['layout_warnings'] == [], 'Actual three-pass layout gate is incomplete')
    require(visual['status'] == 'PASS' and visual['pdf_sha256'] == pdf_hash
            and visual.get('source_sha256', visual.get('canonical_sha256')) == md_hash
            and visual['blocking_findings'] == [], 'Actual final visual receipt is not PASS')
    actual_pages = visual['root_full_pages_actually_viewed']
    require(isinstance(actual_pages, list) and actual_pages and visual['root_review_method'],
            'Actual scoped full-page visual observations are missing')
    pages = visual['page_count']
    require(isinstance(pages, int) and pages > 0 and len(actual_pages) == len(set(actual_pages))
            and all(isinstance(p, int) and 1 <= p <= pages for p in actual_pages), 'Visual page list is invalid')
    require(render['pdf_sha256'] == pdf_hash and render['canonical_source_sha256'] == md_hash
            and render['page_count'] == pages, 'Render/page-count identity differs')
    collaborator = receipts['collaborator']
    require(collaborator['status'] == 'PASS_SCOPED_ACTUAL_FULL_PAGE_REVIEW'
            and collaborator['pdf']['sha256'] == pdf_hash and collaborator['canonical_source']['sha256'] == md_hash
            and collaborator['full_document_pages'] == pages and collaborator['blocking_findings'] == [],
            'Actual collaborator visual receipt does not bind final canonical bytes')
    collaborator_pages = collaborator['actually_viewed_physical_pages']
    require(collaborator_pages == [*range(168, 190), 213, 214, 229]
            and collaborator['full_pages_viewed'] == len(collaborator_pages) == 25,
            'Collaborator actual requested full-page coverage differs')
    adjustments = receipts['canonical_adjustments']
    require(adjustments['schema'] == 'ueot_physics.canonical_layout_and_publication_identity.v3.4'
            and adjustments['initial_integrated_source_sha256'] == integration['canonical_source_sha256']
            and adjustments['current_source_sha256'] == md_hash
            and adjustments['all_three_new_chapter_bodies_preserved'] is True
            and adjustments['embedded_program_count'] == 3
            and adjustments['embedded_program_bytes_unchanged'] is True
            and adjustments['old_programs_reexecuted'] is False,
            'Actual additive canonical layout/publication-identity bridge does not bind final source')
    require(integration['inherited_results_preserved'] == 131
            and integration['new_result_ids'] == [x['id'] for x in claims[131:]]
            and integration['source_set_sha256'] == SOURCE_SET, 'Actual canonical integration receipt differs')
    require(integration['old_757_checks_reexecuted'] is False
            and integration['whole_book_formalized'] is False
            and integration['new_experimental_data'] is False, 'Canonical integration scope widened')

    # Preserve original chapter binding artifacts as exact history, while current statements
    # are freshly checked against the final actual unified public and witness audit receipts.
    normalized = {}
    binding_ids = {}
    all_bound = set()
    for chapter in [60, 61, 62]:
        path = external_binding(chapter)
        binding_raw = read_abs(path)
        binding = json.loads(binding_raw)
        require(binding['chapter'] == chapter, 'Binding chapter mismatch')
        binding_ids[chapter] = planned(NEW / f'chapter{chapter}-bindings.json', binding_raw)
        chapter_input = next(x for x in integration['chapter_inputs'] if x['chapter'] == chapter)
        draft_identity = binding.get('chapter_source', binding.get('chapter_draft'))
        require(draft_identity['sha256'] == chapter_input['sha256'], 'Binding and installed draft input differ')
        draft_raw = read_abs(Path(chapter_input['path']))
        require(digest(draft_raw) == chapter_input['sha256'] and len(draft_raw) == chapter_input['bytes'],
                'Author chapter draft bytes changed after integration')
        require(slices[f'ch-{chapter}'].strip() == draft_raw.decode().strip(),
                'Installed chapter differs from reviewed complete draft: ' + str(chapter))
        rows = binding.get('results', binding.get('numbered_results'))
        require([x['id'] for x in rows] == [x['id'] for x in claims if x['chapter'] == chapter],
                'Binding numbered results differ from actual chapter')
        normalized[chapter] = []
        for result in rows:
            main_names = result.get('public_main_declarations', result.get('main_declarations', []))
            support_names = (result.get('public_support_declarations', result.get('support_declarations', []))
                             + result.get('inherited_declarations', []))
            control_names = result.get('concrete_witness_declarations', result.get('witness_declarations', []))
            names = list(dict.fromkeys(main_names + support_names + control_names))
            require(names and set(names) <= generic_names | witness_names, 'Named claim not in actual complete audit: ' + result['id'])
            require(set(names) <= declaration_source.keys(), 'Named claim source location missing: ' + result['id'])
            require(result['anchor'] == result['id'].lower() and by_id[result['id']]['chapter'] == chapter,
                    'Claim anchor/placement mismatch')
            assumptions = result.get('premises', result.get('assumptions'))
            quantifiers = result.get('quantified_scope', result.get('quantifiers'))
            conclusion = result.get('actual_conclusion', result.get('conclusion'))
            require(assumptions and quantifiers and conclusion and result['open_boundaries'] and result['proof_outline'],
                    'Incomplete scientific responsibility: ' + result['id'])
            grouped = defaultdict(list)
            for name in names:
                grouped[declaration_source[name]].append(name)
            bound_sources = []
            for source, declarations in sorted(grouped.items()):
                is_witness = '/scripts/witnesses/' in source
                bound_sources.append({**bound_source_identity(source), 'declarations': declarations,
                                      'role': 'CONCRETE_WITNESS' if is_witness else 'GENERIC_MODULE',
                                      'actual_named_audit_receipt': str(KERNEL / ('finite-witnesses/verification.json' if is_witness else 'proof-axioms.json')),
                                      'all_names_present_in_actual_named_audit': True})
            negative = result['id'].startswith('N-')
            normalized[chapter].append({
                'id': result['id'], 'title': by_id[result['id']]['title'], 'anchor': result['anchor'],
                'origin': 'NEW_IN_3_4', 'argument_status': ['N' if negative else 'P'],
                'dependency_class': ['B', 'P', 'O'], 'verification_kind': ['analytic', 'lean'] + (['exact_arithmetic'] if negative else []),
                'type_responsibility': {'domain_assumptions': assumptions, 'quantifier_scope': quantifiers,
                                        'proved_conclusion_scope': conclusion,
                                        'roles': [{'class': 'B', 'responsibility': 'Explicit finite state/effect/process interfaces.'},
                                                  {'class': 'P', 'responsibility': 'Declared finite complex mathematical model; not uniquely selected by Core.'},
                                                  {'class': 'O', 'responsibility': 'Actually proved probability, state, marginal, or record comparison.'}]},
                'source_refs': [{'id': 'QM318', 'locator': x['path']} for x in bound_sources],
                'formal': {'source_bindings': bound_sources, 'main_statement_declarations': main_names,
                           'supporting_declarations': support_names, 'supporting_control_declarations': control_names,
                           'generic_declarations': sorted(set(names) & generic_names),
                           'witness_declarations': sorted(set(names) & witness_names),
                           'all_named_declarations_located_in_actual_source': True,
                           'all_names_bound_to_actual_named_audit': True,
                           'unified_acceptance_status': 'PASS_ACTUAL_LOCAL_VERIFICATION',
                           'remote_acceptance_status': 'PENDING_ACTUAL_SOURCE_AND_MAIN_CI',
                           'whole_chapter_formalized': False,
                           'declaration_roles_come_from_actual_audits_not_field_labels': True},
