"""Build a CJK-plus-Latin publication font from explicitly bound local fonts."""
from pathlib import Path
import hashlib
import json
from datetime import datetime, timezone
from fontTools.ttLib import TTFont
from fontTools.ttLib.scaleUpem import scale_upem
from fontTools.merge import Merger
from fontTools import version as fonttools_version

OUT = Path('/tmp/qm319_publication_font')
LATIN = Path('/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf')
CJK = Path('/workspace/scratch/474cf6b1f047/output/pdf/ueot_physics_v33/UEOT_CJK.ttf')
BOOK = Path('/workspace/scratch/f11b4261e187/qm_v318_work/physics/qm/book')
def identity(p):
    return {'path':str(p), 'bytes':p.stat().st_size, 'sha256':hashlib.sha256(p.read_bytes()).hexdigest()}

assert not OUT.exists(), 'Do not overwrite previous font preparation'
assert hashlib.sha256(CJK.read_bytes()).hexdigest() == 'ee38813ea00c3e32add4268fff7fff9e39417b4913cb13be2415164a47807cc2'
OUT.mkdir()
droid = TTFont(CJK)
old_upem = droid['head'].unitsPerEm
scale_upem(droid, 2048)
# This book is horizontal; only the CJK input has vertical metrics.
for tag in ('vhea', 'vmtx', 'VORG'):
    if tag in droid:
        del droid[tag]
scaled = OUT/'DroidSansFallbackFull-2048.ttf'
droid.save(scaled)
latin = TTFont(LATIN)
# XeLaTeX uses its separate Unicode math font; the merger does not implement MATH.
if 'MATH' in latin:
    del latin['MATH']
latin_text = OUT/'DejaVuSans-text.ttf'
latin.save(latin_text)
font = Merger().merge([str(latin_text),str(scaled)])
for n in font['name'].names:
    if n.nameID in (1,4,6):
        n.string = ('UEOTPublication' if n.nameID == 6 else 'UEOT Publication').encode(n.getEncoding())
target = OUT/'UEOTPublication.ttf'
font.save(target)
merged = TTFont(target)
cmap = merged.getBestCmap()
required = {ord(c) for c in (BOOK/'UEOT_PHYSICS.md').read_text() if not c.isspace()}
missing = sorted(required - set(cmap))
# Math-mode-only characters are typeset by the separately selected math font.
report = {'schema':'ueot_physics.publication_font.v3.5','recorded_at_utc':datetime.now(timezone.utc).isoformat(),
    'method':'Scale CJK outline coordinates to the Latin font units-per-em; omit vertical-only CJK metrics for this horizontal book and Latin MATH metadata because XeLaTeX uses its separate math font; merge Latin first; rename output family.',
    'fonttools_version':fonttools_version,'inputs':[identity(LATIN),identity(CJK)],
    'cjk_original_units_per_em':old_upem,'output_units_per_em':merged['head'].unitsPerEm,
    'output':identity(target),'unicode_codepoints':len(cmap),'missing_markdown_codepoints':[
        {'codepoint':f'U+{c:04X}','character':chr(c)} for c in missing],
    'latin_ascii_fully_covered':set(range(33,127))<=set(cmap),'old_v3_3_cjk_font_bytes_reused':True,'same_as_v3_4_cjk_input':False,'recovery_reason':'Use the available byte-bound font from the actual prior v3.3 publication; the v3.4 temporary CJK input is absent. The new merged font and full new PDF layout have separate identities and actual validation.',
    'layout_acceptance':'PENDING_ACTUAL_XELATEX_AND_VISUAL_REVIEW'}
(OUT/'font-identity.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(report,ensure_ascii=False,indent=2))
