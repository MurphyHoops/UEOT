VERIFIER = PKG / 'scripts/verify_finite_witnesses.py'
spec = importlib.util.spec_from_file_location('retained_witness_verifier', VERIFIER)
verifier = importlib.util.module_from_spec(spec)
spec.loader.exec_module(verifier)

def identity(path):
    data = path.read_bytes()
    try:
        name = str(path.relative_to(ROOT))
    except ValueError:
        name = str(path)
    return {'path': name, 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def write(name, data):
    path = OUT / name
    if path.exists():
        raise RuntimeError(f'refusing to overwrite an existing acceptance artifact: {path}')
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    return identity(path)

def checked_receipt(name, source, olean=None):
    path = OUT / name
    receipt = json.loads(path.read_text())
    assert receipt['status'] == 'PASS' and receipt['exit_code'] == 0
    assert receipt['source_unchanged_during_compilation'] is True
    source_identity = identity(source)
    assert receipt['source_sha256'] == source_identity['sha256']
    assert receipt['source_bytes'] == source_identity['bytes']
    log = identity(OUT / receipt['log'])
    assert receipt['log_sha256'] == log['sha256'] and receipt['log_bytes'] == log['bytes']
    assert receipt['shared_lock'] == '/tmp/qm317-lean.lock'
    assert receipt['full_package_rebuild'] is False
    result = {'receipt': identity(path), 'source': source_identity, 'log': log,
              'exit_code': receipt['exit_code'], 'source_matches_successful_invocation': True}
    if olean:
        result['olean'] = identity(olean)
        assert receipt['olean_sha256'] == result['olean']['sha256']
    return result

inventory_path = OUT / 'public-declarations.json'
inventory = json.loads(inventory_path.read_text())
public = inventory['declarations']
sources = sorted({d['source'] for d in public})
for relative in sources:
    path = ROOT / relative
    expected = [d for d in public if d['source'] == relative]
    code = verifier.lean_code(path.read_text())
    namespace = re.findall(r'^namespace\s+([^\n]+)$', code, re.M)
    assert namespace == ['UEOTQM.' + path.stem]
    actual = []
    for line in code.splitlines():
        line = line.strip()
        if match := re.match(r'(?:@\[[^]]+\]\s*)?(?:noncomputable\s+)?(theorem|def|abbrev)\s+(' + verifier.IDENTIFIER + r')(?=\s|:|\()', line):
            actual.append({'kind': match[1], 'name': namespace[0] + '.' + match[2]})
        elif re.match(r'(?:private\s+|protected\s+)?(?:theorem|def|abbrev)\b', line):
            raise ValueError(f'unsupported declaration: {line}')
    assert actual == [{'kind': d['kind'], 'name': d['name']} for d in expected]
    assert all(d['source_sha256'] == identity(path)['sha256'] for d in expected)
    assert not re.search(r'\b(?:sorry|admit|native_decide|axiom)\b', code)

witness_path = PKG / 'scripts/witnesses/QuantumSeparationExamples.lean'

assert not re.search(r'\b(?:sorry|admit|native_decide|axiom)\b', verifier.lean_code(witness_path.read_text()))
audits = {}
for name, logfile, declarations in (
    ('public', '10_PublicAxioms.log', public),
    ('witness', '09_QuantumSeparationExamples.log', witness),
):
    log = OUT / logfile
    audit = verifier.audit_result(log, declarations)
    output_names = re.findall(r"^'([^\n]+)' (?:depends on axioms: \[([^]]*)\]|does not depend on any axioms)", log.read_text(), re.M)
    audit['source_order_exact'] = [name for name, _ in output_names] == [d['name'] for d in declarations]
    assert audit['status'] == 'PASS' and audit['source_order_exact']
    audit['log'] = identity(log)
    audits[name] = write(f'{name}-named-axiom-audit.json', audit)
