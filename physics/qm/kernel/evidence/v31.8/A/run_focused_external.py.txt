#!/usr/bin/env python3
"""Record one locked focused compilation of an A-owned v31.8 Lean source."""
import argparse
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

parser = argparse.ArgumentParser()
parser.add_argument('target', choices=['QuantumSeparation', 'ComplexLocalTomography', 'QuantumSeparationExamples', 'PublicAxioms'])
parser.add_argument('attempt')
args = parser.parse_args()
repo = Path('/workspace/scratch/f11b4261e187/qm_v318_work')
project = repo / 'formalization/ueot-qm'
evidence = repo / 'physics/qm/kernel/evidence/v31.8/A'
witness = args.target.endswith('Examples')
audit = args.target == 'PublicAxioms'
relative = ('scripts/witnesses/' if witness else 'UEOTQM/') + args.target + '.lean'
if audit:
    relative = str(repo.parent / 'handoff/qm318_A_PublicAxioms.lean')
source = project / relative
sha = lambda b: hashlib.sha256(b).hexdigest()
before = source.read_bytes()
command = ['flock', '/tmp/qm317-lean.lock', '/tmp/lean-4.33.1-linux/bin/lake', 'env', 'lean', relative]
olean = project / '.lake/build/lib/lean/UEOTQM' / (args.target + '.olean')
if not witness and not audit:
    command += ['-o', str(olean)]
log = evidence / (args.attempt + '_' + args.target + '.log')
start = datetime.now(timezone.utc).isoformat()
environment = os.environ.copy()
environment['PATH'] = '/tmp/lean-4.33.1-linux/bin:' + environment['PATH']
with log.open('wb') as stream:
    result = subprocess.run(command, cwd=project, env=environment,
        stdout=stream, stderr=subprocess.STDOUT)
end = datetime.now(timezone.utc).isoformat()
receipt = {'status': 'PASS' if result.returncode == 0 else 'FAILED_FOCUSED_ATTEMPT',
    'command': command, 'cwd': str(project), 'started_at_utc': start, 'finished_at_utc': end,
    'exit_code': result.returncode, 'source': relative, 'source_sha256': sha(before),
    'source_bytes': len(before), 'source_unchanged_during_compilation': source.read_bytes() == before,
    'log': log.name, 'log_sha256': sha(log.read_bytes()), 'log_bytes': log.stat().st_size,
    'full_package_rebuild': False, 'shared_lock': '/tmp/qm317-lean.lock'}
if result.returncode == 0 and not witness and not audit:
    receipt['olean_sha256'] = sha(olean.read_bytes())
(evidence / (args.attempt + '_' + args.target + '.json')).write_text(json.dumps(receipt, indent=2) + '\n')
print(json.dumps(receipt))
sys.exit(result.returncode)
