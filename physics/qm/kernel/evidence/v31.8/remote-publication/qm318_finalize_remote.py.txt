class Refused(Exception):
    pass


def require(ok, message):
    if not ok:
        raise Refused(message)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def git_blob(raw):
    return hashlib.sha1(b"blob " + str(len(raw)).encode() + b"\0" + raw).hexdigest()


def encoded(obj):
    return (json.dumps(obj, ensure_ascii=False, indent=2) + "\n").encode()


class Plan:
    def __init__(self, root):
        self.root, self.inputs, self.outputs = root, {}, {}

    def raw(self, path):
        path = Path(path).resolve()
        data = path.read_bytes()
        prior = self.inputs.setdefault(path, data)
        require(prior == data, "Input changed during preflight: " + str(path))
        return data

    def read(self, path):
        return json.loads(self.raw(path))

    def repo(self, value):
        path = Path(value)
        require(not path.is_absolute() and ".." not in path.parts, "Unsafe repository path: " + str(value))
        result = (self.root / path).resolve()
        require(result.is_relative_to(self.root), "Repository path escapes root: " + str(value))
        return result

    def identity(self, value):
        path = self.repo(value)
        raw = self.raw(path)
        return {"path": str(Path(value)), "sha256": sha(raw), "bytes": len(raw)}

    def check_identity(self, record):
        require(self.identity(record["path"]) == {k: record[k] for k in ("path", "sha256", "bytes")},
                "Identity mismatch: " + record["path"])

    def stage(self, rel, value):
        path = self.repo(rel)
        require(path == self.root / rel and not path.is_symlink(), "Metadata target must be a regular repository path: " + str(rel))
        if path not in self.inputs:
            self.inputs[path] = path.read_bytes() if path.exists() else None
        self.outputs[path] = value if isinstance(value, bytes) else encoded(value)

    def unchanged(self):
        for path, raw in self.inputs.items():
            require((path.read_bytes() if path.exists() else None) == raw,
                    "Input/destination changed: " + str(path))


def external(context_path, c, key, default):
    path = Path(c.get(key, default))
    return (path if path.is_absolute() else context_path.parent / path).resolve()


def validate_context(plan, path):
    c = plan.read(path)
    require(c["schema"] == "ueot_qm.remote_finalization_context.v31.8" and c["repository"] == REPOSITORY,
            "Wrong remote context schema/repository")
    require(c["pr_status"] == c["main_status"] == "PASS", "Actual PR and independent main CI must both be PASS")
    require(c["source_count"] == 89 and c["source_set_sha256"] == PUBLICATION_SET,
            "Context source_set must identify final publication inputs, not the original local build")
    for key in ("source_commit", "source_tree", "tested_main_commit", "tested_main_tree", "parent_commit", "parent_tree"):
        require(isinstance(c[key], str) and re.fullmatch(r"[0-9a-f]{40}", c[key]), "Missing actual full Git SHA: " + key)
    for key in ("pull_request", "pr_run", "main_run"):
        require(type(c[key]) is int and c[key] > 0, "Missing actual positive integer: " + key)
    require(c["pr_run"] != c["main_run"], "PR/main runs must be independent actual events")
    require(c["source_branch"].startswith("qm/v31.8-"), "Wrong source branch")
    for key, expected in {
        "pull_request_url": f"https://github.com/{REPOSITORY}/pull/{c['pull_request']}",
        "pr_run_url": f"https://github.com/{REPOSITORY}/actions/runs/{c['pr_run']}",
        "main_run_url": f"https://github.com/{REPOSITORY}/actions/runs/{c['main_run']}",
    }.items():
        require(c[key] == expected, "Actual URL mismatch: " + key)
    stamp = datetime.fromisoformat(c["recorded_at_utc"].replace("Z", "+00:00"))
    require(stamp.utcoffset() is not None and stamp.utcoffset().total_seconds() == 0,
            "An actual stable UTC context timestamp is required")
    return c


def check_source_bridge(plan):
    source = plan.read(plan.repo(E / "source-manifest.json"))
    publication = plan.read(plan.repo(E / "publication-source-manifest.json"))
    proof = plan.read(plan.repo(E / "lean-verification.json"))
    bridge_path = E / "publication-source-reconciliation.json"
    bridge = plan.read(plan.repo(bridge_path))
    require(sha(plan.raw(plan.repo(bridge_path))) == BRIDGE_SHA and bridge["status"] == "PASS_COMMENT_ONLY",
            "Original actual publication bridge changed")
    require(proof["status"] == "PASS" and proof["actual_build_jobs"] == 8799 and
            proof["source_set_sha256"] == UNIFIED_SET and proof["published_research_baseline"] == BASELINE and
            proof["full_scientific_theory_verified"] is False, "Original actual proof identity/scope changed")
    for manifest, expected in ((source, UNIFIED_SET), (publication, PUBLICATION_SET)):
        require(manifest["source_count"] == len(manifest["sources"]) == 89 and
                len({s["path"] for s in manifest["sources"]}) == 89 and manifest["source_set_sha256"] == expected and
                sha(json.dumps(manifest["sources"], sort_keys=True, separators=(",", ":")).encode()) == expected,
                "Wrong source manifest identity")
    uid, pid = bridge["unified_source_identity"], bridge["publication_source_identity"]
    require(uid["source_set_sha256"] == UNIFIED_SET and uid["actual_build_jobs"] == 8799 and
            pid["source_set_sha256"] == PUBLICATION_SET and uid["source_count"] == pid["source_count"] == 89,
            "Original build/publication bridge roles changed")
    for item in (uid["manifest"], uid["verification"], pid["manifest"]):
        plan.check_identity(item)
    old = {x["path"]: x for x in source["sources"]}
    new = {x["path"]: x for x in publication["sources"]}
    rel = "formalization/ueot-qm/UEOTQM/PurificationUniqueness.lean"
    require(old.keys() == new.keys() and [p for p in old if old[p] != new[p]] == bridge["changed_paths"] == [rel],
            "Source difference exceeds the sole authorized comment correction")
    require(len(bridge["changes"]) == 1 and bridge["unchanged_source_paths"] == 88,
            "Comment bridge change population differs")
    change = bridge["changes"][0]
    require(change["path"] == rel and change["before"] == old[rel] and change["after"] == new[rel],
            "Comment bridge rows differ from manifests")
    for key in ("before_snapshot", "after_snapshot", "exact_diff"):
        plan.check_identity(change[key])
    before = plan.raw(plan.repo(change["before_snapshot"]["path"]))
    after = plan.raw(plan.repo(change["after_snapshot"]["path"]))
    old_phrase = b"/-- Different finite auxiliary carriers admit actual isometric extensions into"
    new_phrase = b"/-- Different finite auxiliary carriers have actual isometric extensions into"
    require(before.count(old_phrase) == 1 and before.replace(old_phrase, new_phrase, 1) == after and
            sha(before) == old[rel]["sha256"] and sha(after) == new[rel]["sha256"],
            "Not the exact one-word documentation correction")
    return source, publication, proof, bridge

def actual_scientific_paths(root):
    package = root / "formalization/ueot-qm"
    paths = set((package / "UEOTQM").rglob("*.lean"))
    paths.update(package / n for n in ("UEOTQM.lean", "lakefile.lean", "lake-manifest.json", "lean-toolchain"))
    paths.update(p for p in (package / "scripts").rglob("*") if p.is_file() and p.suffix in {".lean", ".py", ".sh"})
    kernel = root / Q / "kernel"
    paths.update(kernel.rglob("*.py"))
    paths.update(p for p in (kernel / "certificates").rglob("*") if p.is_file() and p.suffix in {".lean", ".json"})
    paths.add(root / ".github/workflows/ueot-qm.yml")
    return sorted(p.relative_to(root).as_posix() for p in paths)


def check_trees_and_publication(plan, context_path, c, publication):
    trees, remote, tree_inputs = {}, {}, {}
    for kind, commit, tree_sha in (("source", c["source_commit"], c["source_tree"]),
                                   ("main", c["tested_main_commit"], c["tested_main_tree"])):
        p = external(context_path, c, kind + "_remote_tree_path", f"qm318_{kind}_remote_tree.json")
        t = plan.read(p)
        require(t["commit"] == commit and t["sha"] == tree_sha and t["truncated"] is False,
                kind + ": wrong/incomplete actual recursive tree readback")
        require(len(t["tree"]) == len({x["path"] for x in t["tree"]}), kind + ": duplicate remote tree path")
        trees[kind] = {x["path"]: x for x in t["tree"] if x["type"] == "blob"}
        remote[kind] = t
        raw = plan.raw(p)
        tree_inputs[kind] = {"path": str(p), "sha256": sha(raw), "bytes": len(raw)}
    require(c["source_tree"] == EXPECTED_SOURCE_TREE,
            "Actual source tree differs from the independently reconstructed frozen publication tree")
    inventory_path = external(context_path, c, "publication_inventory_path", "qm318_publication_inventory.json")
    inventory_raw = plan.raw(inventory_path)
    require(sha(inventory_raw) == INVENTORY_SHA, "Frozen 338-file inventory bytes changed")
    inventory = json.loads(inventory_raw)
    require(inventory["file_count"] == len(inventory["files"]) == len({x["path"] for x in inventory["files"]}) == 338 and
            inventory["source_set_sha256"] == PUBLICATION_SET and inventory["original_unified_source_set_sha256"] == UNIFIED_SET,
            "Wrong complete publication inventory")
    require(inventory["baseline_commit"] == c["parent_commit"] == BASELINE and
            inventory["baseline_tree"] == c["parent_tree"], "Actual published parent differs from frozen payload baseline")
    file_records = {}
    for entry in inventory["files"]:
        rel = entry["path"]
        require(rel.startswith(("physics/qm/", "formalization/ueot-qm/")), "Publication scope escaped QM: " + rel)
        raw = plan.raw(plan.repo(rel))
        require(sha(raw) == entry["sha256"] and len(raw) == entry["bytes"] and git_blob(raw) == entry["git_blob_sha"],
                "Local publication checkpoint differs from frozen bytes: " + rel)
        require(all(t[rel]["sha"] == entry["git_blob_sha"] and t[rel]["mode"] == entry["mode"] for t in trees.values()),
                "Actual source/main publication blob or mode mismatch: " + rel)
        file_records[rel] = {k: entry[k] for k in ("path", "sha256", "bytes", "git_blob_sha", "mode")}
    require(actual_scientific_paths(plan.root) == sorted(s["path"] for s in publication["sources"]),
            "Current scientific input population differs from the frozen 89 publication paths")
    matched = []
    for item in publication["sources"]:
        plan.check_identity(item)
        raw = plan.raw(plan.repo(item["path"]))
        b = git_blob(raw)
        require(all(t[item["path"]]["sha"] == b for t in trees.values()), "Actual CI-source byte mismatch: " + item["path"])
        matched.append({**item, "git_blob_sha": b, "source_and_main_match": True})
    core = plan.read(plan.repo(E / "core-import-closure.json"))
    require(core["status"] == "SOURCE_AND_BUILT_IDENTITY_PASS" and core["core_sources_modified"] is False and
            len(core["modules"]) == len({x["path"] for x in core["modules"]}) == 56, "Actual Core import closure differs")
    core_matches = []
    for item in core["modules"]:
        raw = plan.raw(plan.repo(item["path"]))
        b = git_blob(raw)
        require(sha(raw) == item["source_sha256"] and all(t[item["path"]]["sha"] == b for t in trees.values()),
                "Frozen Core source differs in actual source/main: " + item["path"])
        core_matches.append({"path": item["path"], "sha256": sha(raw), "bytes": len(raw), "git_blob_sha": b,
                             "source_and_main_match": True})
    gates = plan.read(plan.repo(BE / "local-release-gates.json"))
    require(gates["status"] == LOCAL and gates["source_set_sha256"] == UNIFIED_SET and
            gates["publication_source_set_sha256"] == PUBLICATION_SET and gates["counts"]["pages"] == 229,
            "Immutable local book gates differ")
    book, bookfiles = {}, []
    for name, key, gate_key in (("UEOT_PHYSICS.md", "markdown_sha256", "canonical"),
                               ("UEOT_PHYSICS.pdf", "pdf_sha256", "pdf"),
                               ("UEOT_PHYSICS.tex", "tex_sha256", "latex")):
        rel = Q / "book" / name
        raw = plan.raw(plan.repo(rel))
        b = git_blob(raw)
        require(sha(raw) == CANONICAL[name] == gates[gate_key]["sha256"] and
                all(t[str(rel)]["sha"] == b for t in trees.values()), "Canonical book source/main mismatch: " + name)
        book[key] = sha(raw)
        bookfiles.append({"path": str(rel), "sha256": sha(raw), "bytes": len(raw), "git_blob_sha": b,
                          "source_and_main_match": True})
    book.update(edition="3.4", pages=229, numbered_chapters=63, appendices=6, numbered_results=147,
                sections=320, anchors=216)
    if "book" in c:
        require(all(c["book"][key] == book[key] for key in ("edition", "pages", "markdown_sha256", "pdf_sha256", "tex_sha256")),
                "Actual context book identities differ")
    checkpoint = {"source_commit": c["source_commit"], "source_tree": c["source_tree"],
                  "tested_main_commit": c["tested_main_commit"], "tested_main_tree": c["tested_main_tree"],
                  "files": [file_records[str(p)] for p in (*STATE_FILES, *DOCUMENTS, *BOOK_METADATA)],
                  "scope": "Exact local-publication metadata bytes remain recoverable from the actual immutable scientific source commit and these Git blobs. Original local gates referring to mutable metadata paths must be interpreted at that source commit; current remote metadata is a later separate change."}
    return {"inventory": inventory, "inventory_raw": inventory_raw, "remote": remote, "tree_inputs": tree_inputs, "sources": matched,
            "core": core_matches, "book": book, "bookfiles": bookfiles, "gates": gates, "checkpoint": checkpoint}

def protect_history_and_scope(plan, context_path, c):
    baseline = external(context_path, c, "baseline_root", "../qm_v317_work")
    require(baseline.is_dir() and baseline != plan.root, "Read-only closed v31.7 baseline is required")
    historical = set()
    for prefix in ("physics/qm/kernel/evidence", "physics/qm/book/evidence", "physics/qm/status", "physics/qm/audits"):
        historical.update(p for p in (baseline / prefix).rglob("*") if p.is_file())
    for pattern in ("UEOT_QM_KERNEL_v*.md", "UEOT_QM_EXECUTION_v*.md"):
        historical.update((baseline / Q).glob(pattern))
    count = 0
    for old in sorted(historical):
        rel = old.relative_to(baseline)
        if rel in (*STATE_FILES, *DOCUMENTS):
            continue
        require(plan.raw(plan.repo(rel)) == plan.raw(old), "Historical evidence/report changed: " + str(rel))
        count += 1
    require(count == 820, "Historical preservation population differs from the accepted local checkpoint")
    evidence = set()
    for prefix in (Q / "kernel/evidence", Q / "book/evidence"):
        for p in (plan.root / prefix).rglob("*"):
            if p.is_file():
                plan.raw(p)
                evidence.add(p.resolve())
    for rel in NEW_RECEIPTS:
        require(not plan.repo(rel).exists(), "Refusing to overwrite an existing actual remote receipt: " + str(rel))
    workflow = plan.raw(plan.repo(".github/workflows/ueot-qm.yml")).decode()
    on_section = workflow.split("\npermissions:", 1)[0]
    patterns = re.findall(r'^\s+- "([^"]+)"\s*$', on_section, re.M)
    require(len(patterns) == 30 and len(set(patterns)) == 15, "Actual scientific workflow path policy differs")
    targets = tuple(str(p) for p in (*STATE_FILES, *DOCUMENTS, *NEW_RECEIPTS))
    require(not any(fnmatch.fnmatchcase(p, pattern) for p in targets for pattern in patterns),
            "A metadata output would trigger the scientific workflow")
    return {"historical_files_equal_closed_baseline": count, "existing_evidence_files_pinned": len(evidence),
            "metadata_targets_match_scientific_workflow_paths": False}

WRITES = 0


def main():
    global WRITES
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--root", type=Path, required=True)
    parser.add_argument("--context", type=Path, required=True)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument("--apply", action="store_true")
    mode.add_argument("--dry-run", action="store_true")
    mode.add_argument("--plan", action="store_true")
    args = parser.parse_args()
    if args.plan:
        print(json.dumps({"status": "PLAN_ONLY_NOT_ACCEPTANCE", "default_read_only": True,
            "targets": [str(p) for p in (*STATE_FILES, *DOCUMENTS, *NEW_RECEIPTS)],
            "context_schema": "ueot_qm.remote_finalization_context.v31.8",
            "requires": "Actual separate PR/main CI PASS, saved run/jobs/log inputs and complete source/main tree readbacks for all 338 paths; no inferred run or commit identity.",
            "new_scientific_validation": False, "completed_repository_writes": 0}, ensure_ascii=False, indent=2))
        return
    plan, summary = prepare(args.root, args.context)
    if args.apply:
        staged = []
        try:
            plan.unchanged()
            for path, raw in plan.outputs.items():
                with tempfile.NamedTemporaryFile(prefix=".qm318-remote-", dir=path.parent, delete=False) as stream:
                    temp = Path(stream.name)
                    staged.append((temp, path))
                    stream.write(raw)
                    stream.flush()
                    os.fsync(stream.fileno())
                os.chmod(temp, path.stat().st_mode & 0o777 if path.exists() else 0o644)
            # Check every actual input and every destination once more after all
            # temporary outputs are prepared, before the first atomic replacement.
            plan.unchanged()
            for temp, path in staged:
                temp.replace(path)
                WRITES += 1
            for path, raw in plan.outputs.items():
                require(path.read_bytes() == raw, "Actual metadata readback mismatch: " + str(path))
            for path, raw in plan.inputs.items():
                if path not in plan.outputs:
                    require((path.read_bytes() if path.exists() else None) == raw,
                            "Protected input changed during apply: " + str(path))
        finally:
            for temp, _ in staged:
                temp.unlink(missing_ok=True)
        summary.update(status="MAIN_VERIFIED_METADATA_APPLIED", completed_repository_writes=WRITES)
    print(json.dumps(summary, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    try:
        main()
    except (Refused, OSError, ValueError, KeyError, TypeError, UnicodeError) as exc:
        print(json.dumps({"status": "REFUSED_OR_FAILED", "reason": str(exc),
                          "completed_repository_writes": WRITES, "acceptance_not_promoted": WRITES == 0},
                         ensure_ascii=False), file=sys.stderr)
        raise SystemExit(2)
