"""Bind actual v31.8 PR/main acceptance to four current book metadata files.

The default is read-only. --apply is accepted only after the root's actual
MAIN_VERIFIED integration, both independent successful CI verifier receipts,
and the final publication-byte reconciliation exist. No network, Git write,
Lean invocation, runtime test, PDF build, or visual inspection is performed.
The immutable original local build remains bound to a343...; the actual remote
CI input is the separately recorded 054f... publication set. The three updated
metadata files point to the new terminal receipt by path, never by its hash.
"""

REPOSITORY = 'MurphyHoops/UEOT'

    'UEOT_PHYSICS.pdf': '24b61ee3dd8d935afd47386460b352c40d4373e8328e2b4cfcfb8d8c460c60f0',

LOCAL_METADATA = {
    TARGETS[0]: 'c5f953a6f0fe64b304a11c675d236733081d2528cbe6bae28a9aaf9e6f276526',
    TARGETS[1]: 'fcccbbc22050b151981b95ce0b57fa264867a44f70431d35a480f2336b3eb2dc',
    TARGETS[2]: '9de377d0a79bf819739818ee5b8dd3da996325c40c57888c897c8cb7296b6e65',

WRITES = 0


def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def blob_sha(data):
    return hashlib.sha1(b'blob ' + str(len(data)).encode('ascii') + b'\0' + data).hexdigest()


def encode(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode('utf-8')


def source_set(entries):
    return sha(json.dumps(entries, sort_keys=True, separators=(',', ':')).encode('utf-8'))


def prepare(root, context_path):
    root = root.resolve()
    context_path = context_path.resolve()
    protected = {}

    def path(relative):
        p = root / relative
        require(not Path(relative).is_absolute() and p.resolve().is_relative_to(root),
                f'Path escapes repository: {relative}')
        return p

    def take(p):
        p = Path(p).resolve()
        data = p.read_bytes()
        digest = sha(data)
        require(p not in protected or protected[p] == digest,
                f'Input changed during preparation: {p}')
        protected[p] = digest
        return data

    def read(relative):
        return json.loads(take(path(relative)))

    def record(relative, prepared=None):
        data = prepared[relative] if prepared and relative in prepared else take(path(relative))
        return {'path': relative, 'sha256': sha(data), 'bytes': len(data)}

    def check_bound(bound, label):
        require(isinstance(bound, dict) and isinstance(bound.get('path'), str)
                and re.fullmatch('[0-9a-f]{64}', bound.get('sha256', ''))
                and type(bound.get('bytes')) is int and bound['bytes'] >= 0,
                f'Malformed byte binding: {label}')
        p = Path(bound['path'])
        if not p.is_absolute():
            p = path(bound['path'])
        data = take(p)
        require(sha(data) == bound['sha256'] and len(data) == bound['bytes'],
                f'Actual bound input changed: {label}')
        return data

    context_bytes = take(context_path)
    ctx = json.loads(context_bytes)
    require(all(key in ctx for key in CONTEXT_KEYS), 'Missing actual release-context fields.')
    require(ctx['repository'] == REPOSITORY, 'Unexpected repository.')
    require(ctx['pr_status'] == ctx['main_status'] == 'PASS',
            'Both actual PR/main CI PASS receipts are required; pending context refuses writes.')
    require(ctx['source_count'] == 89 and ctx['source_set_sha256'] == PUBLICATION_SET,
            'The actual PR/main context must identify the final 89-source publication set.')
    for key in ('source_commit', 'source_tree', 'tested_main_commit', 'tested_main_tree'):
        require(isinstance(ctx[key], str) and re.fullmatch('[0-9a-f]{40}', ctx[key]),
                f'Missing full actual commit/tree SHA: {key}')
    for key in ('pull_request', 'pr_run', 'main_run'):
        require(type(ctx[key]) is int and ctx[key] > 0, f'Missing actual numeric identity: {key}')
    require(ctx['pr_run'] != ctx['main_run'], 'PR/main must be independent actual workflow runs.')
    now = ctx.get('recorded_at_utc')
    require(isinstance(now, str), 'recorded_at_utc must be supplied by the actual release context.')
    parsed_time = datetime.fromisoformat(now.replace('Z', '+00:00'))
    require(parsed_time.utcoffset() is not None and parsed_time.utcoffset().total_seconds() == 0,
            'The actual metadata timestamp must include UTC; no timestamp is inferred.')

    integration = read(INTEGRATION)
    require(integration.get('schema') == 'ueot_qm.integration_verification.v31.8'
            and integration.get('status') == 'MAIN_VERIFIED', 'Root actual integration is not MAIN_VERIFIED.')
    for key in CONTEXT_KEYS:
        require(integration.get(key) == ctx[key], f'Actual context/integration mismatch: {key}')
    for key in ('tested_inputs_identical', 'core_imports_identical', 'canonical_book_identical',
                'published_paths_identical'):
        require(integration.get(key) is True, f'Root final-byte reconciliation missing: {key}')
    require(integration.get('published_path_count') == 338,
            'The actual frozen 338-path source publication is not reconciled.')
    tree_equal = ctx['source_tree'] == ctx['tested_main_tree']
    require(integration.get('source_and_main_tree_identical') is tree_equal,
            'Whole-tree equality must reflect the actual tree identities.')
    require(integration.get('full_scientific_theory_verified') is False
            and integration.get('empirical_confirmation_added') is False,
            'Remote acceptance must preserve finite formal scope and no new empirical claim.')

    bridge = read(BRIDGE)
    require(sha(take(path(BRIDGE))) == BRIDGE_SHA
            and bridge.get('schema') == 'ueot_qm.publication_source_reconciliation.v31.8'
            and bridge.get('status') == 'PASS_COMMENT_ONLY', 'The immutable one-word source bridge differs.')
    unified_identity = bridge['unified_source_identity']
    publication_identity = bridge['publication_source_identity']
    require(unified_identity['source_count'] == publication_identity['source_count'] == 89
            and unified_identity['source_set_sha256'] == UNIFIED_SET
            and unified_identity['actual_build_jobs'] == 8799
            and publication_identity['source_set_sha256'] == PUBLICATION_SET,
            'Original unified/publication source roles differ.')
    require(integration.get('actual_unified_source_identity') == unified_identity,
            'Root integration must preserve the original actual local build identity.')
    integration_publication = integration.get('publication_source_identity', {})
    for key, value in publication_identity.items():
        require(integration_publication.get(key) == value,
                f'Root publication identity differs: {key}')
    local_proof = json.loads(check_bound(unified_identity['verification'], 'actual original unified build'))
    require(local_proof.get('status') == 'PASS' and local_proof.get('source_set_sha256') == UNIFIED_SET,
            'Original local proof acceptance is absent or rebound to different raw bytes.')
    manifests = {}
    for kind, identity in (('unified', unified_identity), ('publication', publication_identity)):
        manifest = json.loads(check_bound(identity['manifest'], kind + ' source manifest'))
        entries = manifest['sources']
        require(manifest['schema'] == 'ueot_qm.source_manifest.v2'
                and manifest['source_count'] == len(entries) == 89
                and source_set(entries) == manifest['source_set_sha256'] == identity['source_set_sha256'],
                f'Invalid {kind} source aggregate identity.')
        names = [entry['path'] for entry in entries]
        require(names == sorted(set(names)), f'{kind} source paths are not unique and sorted.')
        manifests[kind] = {entry['path']: entry for entry in entries}
    require(set(manifests['unified']) == set(manifests['publication']), 'Source path sets differ.')
    changed = [relative for relative in manifests['publication']
               if manifests['publication'][relative] != manifests['unified'][relative]]
    changed_path = 'formalization/ueot-qm/UEOTQM/PurificationUniqueness.lean'
    require(changed == [changed_path] and bridge['changed_paths'] == changed
            and bridge['unchanged_source_paths'] == 88, 'The bridge is no longer the sole documentation edit.')
    for relative, entry in manifests['publication'].items():
        check_bound(entry, 'current final source ' + relative)
    change = bridge['changes'][0]
    before = check_bound(change['before_snapshot'], 'original documentation snapshot')
    after = check_bound(change['after_snapshot'], 'final documentation snapshot')
    check_bound(change['exact_diff'], 'actual documentation diff')
    require(sha(before) == manifests['unified'][changed_path]['sha256']
            and len(before) == manifests['unified'][changed_path]['bytes']
            and after == take(path(changed_path)), 'The unique changed source snapshots differ.')
    old_word = b'/-- Different finite auxiliary carriers admit actual isometric extensions into'
    new_word = b'/-- Different finite auxiliary carriers have actual isometric extensions into'
    require(before.count(old_word) == 1 and before.replace(old_word, new_word, 1) == after,
            'The exact current source differs beyond the documented word.')

    ci_paths, ci = {}, {}
    ci_counts = {
        'generic_modules': 35, 'explicit_theorems': 747, 'generic_axiom_reports': 985,
        'printed_source_hash_count': 39, 'finite_witness_total_audits': 269,
        'finite_witness_file_count': 10, 'finite_witness_theorem_count': 193,
        'finite_witness_def_count': 71, 'finite_witness_abbrev_count': 5,
        'runtime_test_methods': 100, 'demos': 4, 'positive_instances': 6,
        'positive_instance_axiom_targets': 42, 'corrupted_true_rejections': 4,
        'same_data_false_proofs': 4, 'false_witness_axiom_targets': 4,
        'local_accepted_source_count': 89, 'publication_source_count': 89,
    }
    for kind, commit, event in (('pr', ctx['source_commit'], 'pull_request'),
                                ('main', ctx['tested_main_commit'], 'push')):
        relative = KERNEL + f'ci/{kind}-{ctx[kind + "_run"]}-verification.json'
        require(integration.get(kind + '_verification') == relative,
                f'Root {kind} receipt path differs from the actual run identity.')
        receipt = read(relative)
        require(receipt.get('schema') == 'ueot_qm.actual_ci_verification.v31.8'
                and receipt.get('status') == 'PASS' and receipt.get('verification_complete') is True,
                f'The actual {kind} CI verifier did not complete successfully.')
        require(receipt.get('repository') == REPOSITORY and receipt.get('event') == kind
                and receipt.get('workflow_event') == event and receipt.get('source_commit') == commit
                and receipt.get('run_id') == ctx[kind + '_run'], f'Wrong actual {kind} workflow identity.')
        for key in ('jobs_all_success', 'recorded_checkout_matches_expected',
                    'generic_axiom_names_and_foundation_equal_local', 'all_explicit_theorem_names_audited',
                    'printed_source_hashes_equal_local', 'printed_source_hashes_equal_publication_manifest',
                    'runtime_method_names_equal_local', 'input_files_unchanged_during_verification',
                    'publication_source_bridge_verified', 'publication_sources_unchanged_during_verification',
                    'local_witness_reference_validated', 'dependency_lock_gate_success', 'archive_steps_success'):
            require(receipt.get(key) is True, f'Actual {kind} verification is incomplete: {key}')
        for key, value in ci_counts.items():
            require(receipt.get(key) == value, f'Actual {kind} count mismatch: {key}')
        require(receipt.get('allowed_foundation') == FOUNDATION
                and receipt.get('local_accepted_source_set_sha256') == UNIFIED_SET
                and receipt.get('publication_source_set_sha256') == PUBLICATION_SET
                and receipt.get('publication_source_bridge_sha256') == BRIDGE_SHA,
                f'Actual {kind} foundation or dual-source identity differs.')
        require(receipt.get('full_scientific_theory_verified') is False
                and receipt.get('artifact_zip_downloaded') is False
                and receipt.get('artifact_zip_contents_verified') is False,
                f'Unexpected actual {kind} evidence scope; explicit review is required.')
        require(type(receipt.get('build_jobs')) is int and receipt['build_jobs'] > 0,
                f'Actual {kind} successful build-job count is absent.')
        # These are bindings to already-verified actual run/jobs/logs and references;
        # this helper does not re-execute the CI parser, Lean, or any runtime program.
        for name, bound in receipt['input_files'].items():
            check_bound(bound, kind + ' CI verifier input ' + name)
        ci_paths[kind], ci[kind] = relative, receipt

    closure = read(KERNEL + 'core-import-closure.json')
    require(closure.get('transitive_module_count') == len(closure['modules']) == 56
            and closure.get('core_sources_modified') is False, 'The actual recursive Core closure differs.')
    for entry in closure['modules']:
        require(sha(take(path(entry['path']))) == entry['source_sha256'],
                f'Imported Core source changed: {entry["path"]}')
    for name, digest in CANONICAL.items():
        require(sha(take(path(BOOK + name))) == digest, f'Canonical bytes changed: {name}')
    for key, value in {'edition': EDITION, 'pages': 229,
                       'markdown_sha256': CANONICAL['UEOT_PHYSICS.md'],
                       'pdf_sha256': CANONICAL['UEOT_PHYSICS.pdf'],
                       'tex_sha256': CANONICAL['UEOT_PHYSICS.tex']}.items():
        require(integration.get('book', {}).get(key) == value, f'Root remote book binding differs: {key}')

    inventory_bound = integration['publication_inventory']
    inventory_bytes = check_bound(inventory_bound, 'actual frozen publication inventory')
    inventory = json.loads(inventory_bytes)
    require(inventory.get('schema') == 'ueot_qm.publication_inventory.v31.8'
            and inventory.get('source_count') == 89
            and inventory.get('source_set_sha256') == PUBLICATION_SET
            and inventory.get('original_unified_source_set_sha256') == UNIFIED_SET
            and inventory.get('file_count') == len(inventory['files']) == 338,
            'Root inventory does not identify the actual frozen v31.8 publication.')
    inventory_files = {entry['path']: entry for entry in inventory['files']}
    require(len(inventory_files) == 338, 'Frozen publication inventory contains duplicate paths.')
    if ctx.get('publication_inventory_path'):
        p = Path(ctx['publication_inventory_path'])
        p = p if p.is_absolute() else context_path.parent / p
        require(take(p) == inventory_bytes, 'Context and root integration identify different inventories.')

    local = read(EVIDENCE + 'local-release-gates.json')
    require(sha(take(path(EVIDENCE + 'local-release-gates.json'))) == LOCAL_GATE_SHA
            and local.get('status') == LOCAL_STATE and local.get('remote_status') == 'PENDING'
            and local.get('remote_verified') is False, 'Immutable local book checkpoint changed.')
    for key in ('proof_gate', 'structure_gate', 'pdf_build_gate', 'visual_gate', 'dossier_gate'):
        require(local.get(key) == 'PASS', f'Actual local book gate missing: {key}')
    require(local['unified_source_identity'] == unified_identity
            and local['source_set_sha256'] == UNIFIED_SET
            and local['publication_source_set_sha256'] == PUBLICATION_SET
            and local['counts']['chapters'] == 63 and local['counts']['appendices'] == 6
            and local['counts']['numbered_results'] == 147 and local['counts']['inherited'] == 131
            and local['counts']['new'] == 16 and local['counts']['pages'] == 229,
            'The actual book checkpoint source roles or structure differ.')
    for key in ('canonical', 'pdf', 'latex', 'dossier', 'source_manifest', 'readme',
                'root_actual_local_acceptance', 'actual_source_manifest', 'structure', 'pdf_build',
                'visual_receipt', 'collaborator_visual_receipt', 'render_receipt', 'dossier_review',
                'prior_preservation', 'legacy_source_reuse', 'generator', 'context'):
        check_bound(local[key], 'immutable local gate ' + key)
    visual = read(EVIDENCE + 'pdf-visual-review.json')
    collaborator_visual = read(EVIDENCE + 'pdf-collaborator-review.json')
    require(visual.get('status') == 'PASS' and visual.get('page_count') == 229
            and visual.get('pdf_sha256') == CANONICAL['UEOT_PHYSICS.pdf']
            and visual['all_page_overview']['actually_viewed'] is True
            and visual['all_page_overview']['sheets'] == 7
            and visual['root_full_pages_actually_viewed'] ==
            [1, 4, 40, 151, 152, 153, 170, 180, 186, 199, 205, 209, 213, 214, 229]
            and visual['collaborator_review']['actual_full_pages'] == list(range(168, 190)) + [213, 214, 229]
            and collaborator_visual.get('status') == 'PASS_SCOPED_ACTUAL_FULL_PAGE_REVIEW',
            'Actual inherited visual scope differs; no extra review is inferred.')

    originals = {relative: take(path(relative)) if path(relative).is_file() else None for relative in TARGETS}
    require(originals[REMOTE] is None, 'Remote receipt already exists; reconcile instead of overwriting history.')
    for relative, digest in LOCAL_METADATA.items():
        require(originals[relative] is not None and sha(originals[relative]) == digest,
                f'Frozen current local book metadata changed: {relative}')
    checkpoint = integration['local_metadata_checkpoint']
    require(checkpoint.get('source_commit') == ctx['source_commit']
            and checkpoint.get('source_tree') == ctx['source_tree'],
            'Local metadata checkpoint must identify the actual source commit/tree.')
    checkpoint_files = {entry['path']: entry for entry in checkpoint['files']}
    require(len(checkpoint['files']) == len(checkpoint_files) == 12,
            'Root local checkpoint must bind the 12 pre-remote current metadata files.')
    previous = {}
    for relative, key in zip(TARGETS[:3], ('source_manifest', 'chapter_dossier', 'readme')):
        data = originals[relative]
        inv = inventory_files[relative]
        old = checkpoint_files[relative]
        for bound in (inv, old):
            require(bound['sha256'] == sha(data) and bound['bytes'] == len(data)
                    and bound['git_blob_sha'] == blob_sha(data),
                    f'Actual source-commit local metadata binding differs: {relative}')
        previous[key] = {**record(relative), 'git_blob_sha': blob_sha(data),
                         'source_commit': ctx['source_commit'], 'source_tree': ctx['source_tree'],
                         'historical_resolution': 'Read this path at the actual source commit, not at the later metadata commit.'}
    previous['local_release'] = record(EVIDENCE + 'local-release-gates.json')
    previous['dossier_review'] = record(EVIDENCE + 'dossier-review.json')

    manifest = json.loads(originals[TARGETS[0]])
    dossier = json.loads(originals[TARGETS[1]])
    old_manifest, old_dossier = copy.deepcopy(manifest), copy.deepcopy(dossier)
    require(manifest['current_code_identity']['status'] == LOCAL_STATE
            and dossier['current_release_stage'] == LOCAL_STATE
            and dossier['completed_v31_8_contract']['status'] == LOCAL_STATE,
            'Current book metadata is no longer the accepted pre-remote checkpoint.')
    require(manifest['unified_source_identity'] == dossier['unified_source_identity'] == unified_identity,
            'Current original local source identity differs.')
    evidence_files = sorted(p for p in path(BOOK + 'evidence/').rglob('*') if p.is_file())
    for p in path(BOOK).rglob('*'):
        if p.is_file() and p.relative_to(root).as_posix() not in TARGETS:
            take(p)

    identity = {key: ctx[key] for key in CONTEXT_KEYS}
    identity.update({
        'pull_request_url': f'https://github.com/{REPOSITORY}/pull/{ctx["pull_request"]}',
        'pr_run_url': f'https://github.com/{REPOSITORY}/actions/runs/{ctx["pr_run"]}',
        'main_run_url': f'https://github.com/{REPOSITORY}/actions/runs/{ctx["main_run"]}',
        'source_and_main_tree_identical': tree_equal,
    })
    for key in ('pull_request_url', 'pr_run_url', 'main_run_url'):
        require(ctx.get(key, identity[key]) == identity[key], f'Context URL mismatch: {key}')
    reference = {'path': REMOTE, 'status': 'MAIN_VERIFIED',
                 'hash_binding_direction': 'The external receipt binds updated metadata hashes; no receipt hash is embedded here.'}
    actual_receipts = {'integration': record(INTEGRATION),
                       'pr': record(ci_paths['pr']), 'main': record(ci_paths['main'])}
    remote_source = {
        'edition': EDITION, 'status': 'MAIN_VERIFIED', 'remote_status': 'MAIN_VERIFIED',
        'remote_verified': True, **identity,
        'manifest': copy.deepcopy(publication_identity['manifest']),
        'actual_receipts': actual_receipts, 'publication_reconciliation': record(BRIDGE),
        'identity_rule': 'Actual source PR and independent merged-main CI test the final 89-source publication bytes. The prior a343... local build is a separate immutable input. This later metadata commit is not either CI-tested commit.',
        'metadata_only_followup': True,
    }
    current_publication = copy.deepcopy(manifest['publication_source_identity'])
    current_publication.update({
        'status': 'MAIN_VERIFIED', 'remote_status': 'MAIN_VERIFIED', 'remote_verified': True,
        'actual_remote_identity': copy.deepcopy(remote_source),
        'scope': 'The 054f... final raw publication set is bound to actual independent PR/main CI and root per-path Git-blob reconciliation. The original 8799-job local run remains bound exclusively to the unchanged a343... unified_source_identity.',
    })

    manifest['remote_metadata_completion'] = {
        'status': 'MAIN_VERIFIED', 'remote_verified': True, 'recorded_at_utc': now,
        'receipt': reference, 'prior_metadata': previous['source_manifest'],
        'prior_current_code_identity': copy.deepcopy(old_manifest['current_code_identity']),
        'prior_remote_metadata_completion': copy.deepcopy(old_manifest['remote_metadata_completion']),
        'immutable_local_receipts': [previous['local_release'], previous['dossier_review']],
        'metadata_commit_is_not_ci_tested_source_commit': True,
    }
    manifest['current_release_stage'] = 'MAIN_VERIFIED'
    manifest['current_code_baseline_role'] = (
        'Historical authoring baseline. current_code_identity retains the original local unified input; '
        'current_remote_code_identity records the separately tested final publication source and merged main.')
    # Do not spread the common release identity into this object: its a343... raw
    # manifest and original 8799-job build were never the final remote CI input.
    manifest['current_code_identity'].update({
        'status': 'PASS_ACTUAL_LOCAL_UNIFIED_INPUT', 'remote_verified': False,
        'remote_status': 'NOT_THE_REMOTE_CI_INPUT',
        'identity_rule': 'Immutable successful original local unified input. Its source-set and manifest hashes remain a343... and the original manifest. Final 054f... remote acceptance is recorded separately in current_remote_code_identity; no old local run is relabelled.',
    })
    manifest['current_remote_code_identity'] = copy.deepcopy(remote_source)
    manifest['publication_source_identity'] = copy.deepcopy(current_publication)
    manifest['local_release_receipt_scope'] = (
        'Immutable pre-remote checkpoint, including its original pending status and source-commit metadata bytes; '
        'current metadata hashes are bound by evidence/v3.4/remote-publication.json.')
    prepared = {TARGETS[0]: encode(manifest)}

    dossier['remote_metadata_completion'] = {
        'status': 'MAIN_VERIFIED', 'remote_verified': True, 'recorded_at_utc': now,
        **identity, 'receipt': reference, 'prior_metadata': previous['chapter_dossier'],
        'prior_remote_metadata_completion': copy.deepcopy(old_dossier['remote_metadata_completion']),
        'prior_validation': copy.deepcopy(old_dossier['validation']),
        'prior_completion_boundary': copy.deepcopy(old_dossier['completion_boundary']),
        'prior_release_acceptance': copy.deepcopy(old_dossier['release_acceptance']),
        'prior_local_assembly_status_scope': copy.deepcopy(old_dossier['local_assembly_status_scope']),
        'immutable_local_receipts': [previous['local_release'], previous['dossier_review']],

def main():
    global WRITES
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root', type=Path, required=True)
    parser.add_argument('--context', type=Path, required=True)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument('--apply', action='store_true')
    mode.add_argument('--plan', action='store_true')
    mode.add_argument('--dry-run', action='store_true')
    args = parser.parse_args()
    if args.plan:
        print(json.dumps({
            'status': 'PLAN_ONLY_NOT_EXECUTED', 'targets': list(TARGETS),
            'context_required': list(CONTEXT_KEYS) + ['recorded_at_utc'],
            'requires': 'Actual separate PR/main CI PASS, root MAIN_VERIFIED integration, 338-path reconciliation and 12-file source-commit local metadata checkpoint.',
            'default_read_only': True, 'no_scientific_reexecution': True,
            'completed_repository_writes': 0,
        }, ensure_ascii=False, indent=2))
        return
    prepared, originals, protected, summary = prepare(args.root, args.context)
    if args.apply:
        root = args.root.resolve()
        for p, digest in protected.items():
            require(p.is_file() and sha(p.read_bytes()) == digest, f'Protected input changed before apply: {p}')
        for relative, data in originals.items():
            p = root / relative
            require(p.resolve().is_relative_to(root / 'physics/qm/book'), f'Unsafe target: {relative}')
            require((p.read_bytes() if p.is_file() else None) == data, f'Metadata changed concurrently: {relative}')
        # These four atomic replacements are the only repository mutations.
        # Any failure reports the actual completed write count, without altering
        # the saved local evidence or pretending an incomplete apply succeeded.
        for relative in TARGETS:
            target = root / relative
            temporary = None
            try:
                with tempfile.NamedTemporaryFile(prefix='.' + target.name + '.qm318-',
                                                 dir=target.parent, delete=False) as stream:
                    temporary = Path(stream.name)
                    stream.write(prepared[relative])
                    stream.flush()
                temporary.replace(target)
                temporary = None
                WRITES += 1
            finally:
                if temporary is not None:
                    temporary.unlink(missing_ok=True)
        for relative in TARGETS:
            require((root / relative).read_bytes() == prepared[relative], f'Metadata readback failed: {relative}')
        for p, digest in protected.items():
            require(p.is_file() and sha(p.read_bytes()) == digest, f'Protected bytes changed during apply: {p}')
        summary['status'] = 'MAIN_VERIFIED_BOOK_METADATA_APPLIED'
        summary['completed_repository_writes'] = WRITES
    print(json.dumps(summary, ensure_ascii=False, indent=2))


if __name__ == '__main__':
