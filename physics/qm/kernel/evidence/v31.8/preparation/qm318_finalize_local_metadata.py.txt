class Refused(RuntimeError):
    pass


def require(ok, reason):
    if not ok:
        raise Refused(reason)


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def read(path):
    require(path.is_file(), f"Missing required actual input: {path}")
    return json.loads(path.read_bytes())


def encoded(obj):
    return (json.dumps(obj, ensure_ascii=False, indent=2) + "\n").encode()


def repo_path(root, value):
    p = Path(value)
    p = p if p.is_absolute() else root / p
    p = p.resolve()
    require(p.is_relative_to(root), f"Repository path escapes root: {value}")
    return p


def external(context_dir, value):
    p = Path(value)
    return (p if p.is_absolute() else context_dir / p).resolve()


def identity(root, p):
    raw = p.read_bytes()
    return {"path": p.relative_to(root).as_posix(), "sha256": digest(raw), "bytes": len(raw)}


def check_identity(root, item):
    p = repo_path(root, item["path"])
    raw = p.read_bytes()
    require(digest(raw) == item["sha256"], f"Receipt/file hash mismatch: {p}")
    if "bytes" in item:
        require(len(raw) == item["bytes"], f"Receipt/file byte count mismatch: {p}")
    if "size" in item:
        require(len(raw) == item["size"], f"Receipt/file size mismatch: {p}")
    return p


def original_bytes(context_dir, ctx, rel):
    expected = ctx["originals"][rel]
    require(expected is not None, f"No original content for {rel}")
    p = external(context_dir, ctx["originals_dir"]) / rel
    raw = p.read_bytes()
    require(digest(raw) == expected, f"Original snapshot changed: {rel}")
    return raw


def source_paths(root):
    package = root / "formalization/ueot-qm"
    paths = set((package / "UEOTQM").rglob("*.lean"))
    paths.update(package / n for n in ("UEOTQM.lean", "lakefile.lean", "lake-manifest.json", "lean-toolchain"))
    paths.update(p for p in (package / "scripts").rglob("*") if p.is_file() and p.suffix in {".py", ".lean", ".sh"})
    kernel = root / "physics/qm/kernel"
    paths.update(kernel.rglob("*.py"))
    paths.update(p for p in (kernel / "certificates").rglob("*") if p.is_file() and p.suffix in {".lean", ".json"})
    paths.add(root / ".github/workflows/ueot-qm.yml")
    return sorted(paths)


def publication_bridge(root, e, source, proof):
    """Check the one authorized comment correction without relabelling a build."""
    bridge_path = e / "publication-source-reconciliation.json"
    bridge = read(bridge_path)
    publication = read(e / "publication-source-manifest.json")
    require(bridge["schema"] == "ueot_qm.publication_source_reconciliation.v31.8" and
            bridge["status"] == "PASS_COMMENT_ONLY", "No actual comment-only publication bridge")
    unified_id = bridge["unified_source_identity"]
    publication_id = bridge["publication_source_identity"]
    require(unified_id["source_count"] == publication_id["source_count"] == SOURCE_COUNT,
            "Source bridge count mismatch")
    require(unified_id["source_set_sha256"] == SOURCE_SHA and
            publication_id["source_set_sha256"] == PUBLICATION_SHA,
            "Source bridge identity mismatch")
    require(unified_id["actual_build_jobs"] == proof["actual_build_jobs"] == 8799,
            "Source bridge does not bind the actual original build")
    require(check_identity(root, unified_id["manifest"]) == e / "source-manifest.json" and
            check_identity(root, unified_id["verification"]) == e / "lean-verification.json" and
            check_identity(root, publication_id["manifest"]) == e / "publication-source-manifest.json",
            "Source bridge references unexpected receipts")
    for manifest, expected in ((source, SOURCE_SHA), (publication, PUBLICATION_SHA)):
        require(manifest["source_count"] == len(manifest["sources"]) == SOURCE_COUNT and
                manifest["source_set_sha256"] == expected and
                digest(json.dumps(manifest["sources"], sort_keys=True, separators=(",", ":")).encode()) == expected,
                "Recorded source manifest aggregate mismatch")
    before = {x["path"]: x for x in source["sources"]}
    after = {x["path"]: x for x in publication["sources"]}
    require(len(before) == len(after) == SOURCE_COUNT and before.keys() == after.keys(),
            "Publication changed the actual input population")
    rel = "formalization/ueot-qm/UEOTQM/PurificationUniqueness.lean"
    require([p for p in before if before[p] != after[p]] == bridge["changed_paths"] == [rel] and
            bridge["unchanged_source_paths"] == 88 and len(bridge["changes"]) == 1,
            "Publication change exceeds the one authorized documentation word")
    change = bridge["changes"][0]
    require(change["path"] == rel and change["before"] == before[rel] and change["after"] == after[rel],
            "Publication change records do not match actual manifests")
    old = check_identity(root, change["before_snapshot"]).read_bytes()
    new = check_identity(root, change["after_snapshot"]).read_bytes()
    check_identity(root, change["exact_diff"])
    old_phrase = b"/-- Different finite auxiliary carriers admit actual isometric extensions into"
    new_phrase = b"/-- Different finite auxiliary carriers have actual isometric extensions into"
    require(old.count(old_phrase) == 1 and old.replace(old_phrase, new_phrase, 1) == new and
            digest(old) == before[rel]["sha256"] and digest(new) == after[rel]["sha256"] and
            repo_path(root, rel).read_bytes() == new,
            "Source snapshots differ by more than the authorized comment-only replacement")
    require(change["erased_source_equal"] is True and change["declarations_diagnostics_and_instances_equal"] is True and
            bridge["scientific_claims_changed"] is False and bridge["core_changed"] is False and
            bridge["additional_compiler_invocations"] == 0 and
            bridge["unified_receipts_preserved_original_bytes"] is True and
            bridge["actual_workflow_raw_proof_escape_regex_passed"] is True and
            bridge["all_final_source_bytes_verified"] is True,
            "Publication bridge has not preserved the original scientific scope")
    return publication, bridge, identity(root, bridge_path)


def protect_inputs(root, baseline, publication, core, book_files):
    actual = [identity(root, p) for p in source_paths(root)]
    require(actual == publication["sources"], "Actual source set differs from the reconciled publication bytes")
    aggregate = digest(json.dumps(actual, sort_keys=True, separators=(",", ":")).encode())
    require(aggregate == PUBLICATION_SHA and len(actual) == SOURCE_COUNT, "Frozen 89-source publication identity mismatch")
    require(len(core["modules"]) == 56, "Expected 56 actual recursive Core imports")
    for m in core["modules"]:
        p = repo_path(root, m["path"])
        require(digest(p.read_bytes()) == m["source_sha256"], f"Accepted Core source changed: {m['path']}")
        require((baseline / m["path"]).read_bytes() == p.read_bytes(), f"Historical Core bytes changed: {m['path']}")
    for item in book_files:
        check_identity(root, item)
    historical = set()
    for prefix in ("physics/qm/kernel/evidence", "physics/qm/book/evidence", "physics/qm/status", "physics/qm/audits"):
        historical.update(p for p in (baseline / prefix).rglob("*") if p.is_file())
    for pattern in ("UEOT_QM_KERNEL_v*.md", "UEOT_QM_EXECUTION_v*.md"):
        historical.update((baseline / "physics/qm").glob(pattern))
    count = 0
    for old in sorted(historical):
        rel = old.relative_to(baseline).as_posix()
        if rel in OWNED:
            continue
        current = root / rel
        require(current.is_file() and current.read_bytes() == old.read_bytes(), f"Historical evidence/report changed: {rel}")
        count += 1
    return {"scientific_sources": len(actual), "core_sources": 56, "canonical_book_files": len(book_files), "historical_files": count}


def normalized_binding(entry, chapter):
    return {
        "book_id": entry["id"], "anchor": entry["anchor"], "title": entry["title"], "chapter": chapter,
        "premises": entry.get("premises", entry.get("assumptions")),
        "quantifiers": entry.get("quantified_scope", entry.get("quantifiers")),
        "conclusion": entry.get("actual_conclusion", entry.get("conclusion")),
        "main": entry.get("public_main_declarations", entry.get("main_declarations")),
        "support": list(dict.fromkeys(entry.get("public_support_declarations", entry.get("support_declarations", []))
                                      + entry.get("inherited_declarations", []))),
        "witnesses": entry.get("concrete_witness_declarations", entry.get("witness_declarations", [])),
        "limits": entry["open_boundaries"], "proof_outline": entry.get("proof_outline", []),
    }

def paragraph_replace(text, prefix, replacement):
    require(text.count(prefix) == 1, f"Ambiguous source paragraph: {prefix}")
    start = text.index(prefix)
    end = text.find("\n\n", start)
    end = len(text) if end == -1 else end
    return text[:start] + replacement + text[end:]


def exact_replace(text, old, new):
    require(text.count(old) == 1, f"Ambiguous literal replacement: {old[:90]}")
    return text.replace(old, new, 1)
