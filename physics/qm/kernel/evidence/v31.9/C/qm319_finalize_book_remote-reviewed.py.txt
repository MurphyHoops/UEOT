#!/usr/bin/env python3
"""Bind actual v31.9 PR/main acceptance to four current edition-3.5 book files.

Default: read-only. --apply requires the root's MAIN_VERIFIED integration,
independent successful PR/main CI receipts, actual recursive Git-tree readbacks,
and complete frozen-publication reconciliation. No remote identity is inferred.
The same 98 raw scientific inputs bind local acceptance and remote publication.
Local execution of three new witnesses remains separate from ten historically
reused witnesses/certificates/runtime/demos, even when CI later executes them.
No network, Git write, Lean, runtime, typesetting, or visual inspection is used.
Three updated metadata files refer to the terminal receipt by path, not hash.
"""
from __future__ import annotations

import argparse
import copy
from datetime import datetime
import hashlib
import json
import os
from pathlib import Path
import re
import sys
import tempfile


REPOSITORY = 'MurphyHoops/UEOT'
EDITION = '3.5'
LOCAL_STATE = 'LOCAL_VERIFIED_REMOTE_PENDING'
SOURCE_SET = 'bf026d2a1f7d8c8bd1f9f6c1ad9015aa20bdae6bd36b9d432c7ce6f0581effc0'
SOURCE_COUNT = 98
BOOK = 'physics/qm/book/'
EVIDENCE = BOOK + 'evidence/v3.5/'
KERNEL = 'physics/qm/kernel/evidence/v31.9/'
REMOTE = EVIDENCE + 'remote-publication.json'
INTEGRATION = KERNEL + 'integration-verification.json'
LOCAL_GATE = EVIDENCE + 'local-release-gates.json'
TARGETS = (BOOK + 'SOURCE_MANIFEST.json', BOOK + 'CHAPTER_DOSSIER.json',
           BOOK + 'README.md', REMOTE)
CANONICAL_KEYS = {'canonical': ('UEOT_PHYSICS.md', 'markdown_sha256'),
                  'pdf': ('UEOT_PHYSICS.pdf', 'pdf_sha256'),
                  'latex': ('UEOT_PHYSICS.tex', 'tex_sha256')}
CONTEXT_KEYS = ('repository', 'source_commit', 'source_tree', 'tested_main_commit',
                'tested_main_tree', 'pull_request', 'pr_run', 'main_run',
                'pr_status', 'main_status', 'source_count', 'source_set_sha256')
FOUNDATION = ['Classical.choice', 'Quot.sound', 'propext']
WRITES = 0


def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def blob_sha(data):
    return hashlib.sha1(b'blob ' + str(len(data)).encode('ascii') + b'\0' + data).hexdigest()


def encode(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode('utf-8')


def source_set(entries):
    return sha(json.dumps(entries, sort_keys=True, separators=(',', ':')).encode('utf-8'))


def prepare(root, context_path):
    root, context_path = root.resolve(), context_path.resolve()
    protected = {}

    def path(relative):
        p = root / relative
        require(not Path(relative).is_absolute() and p.resolve().is_relative_to(root),
                f'Path escapes repository: {relative}')
        return p

    def take(p):
        p = Path(p).resolve()
        data = p.read_bytes()
        require(p not in protected or protected[p] == sha(data),
                f'Input changed during preparation: {p}')
        protected[p] = sha(data)
        return data

    def read(relative):
        return json.loads(take(path(relative)))

    def record(relative, prepared=None):
        data = prepared[relative] if prepared and relative in prepared else take(path(relative))
        return {'path': relative, 'sha256': sha(data), 'bytes': len(data)}

    def check_bound(bound, label):
        require(isinstance(bound, dict) and isinstance(bound.get('path'), str)
                and re.fullmatch('[0-9a-f]{64}', bound.get('sha256', ''))
                and type(bound.get('bytes')) is int and bound['bytes'] >= 0,
                f'Malformed byte binding: {label}')
        p = Path(bound['path'])
        data = take(p if p.is_absolute() else path(bound['path']))
        require(sha(data) == bound['sha256'] and len(data) == bound['bytes'],
                f'Actual bound input changed: {label}')
        return data

    def rows_by_path(rows, expected, label):
        require(isinstance(rows, list) and len(rows) == expected, f'Wrong row count: {label}')
        out = {entry['path']: entry for entry in rows}
        require(len(out) == len(rows), f'Duplicate paths: {label}')
        for relative in out:
            path(relative)
        return out

    context_bytes = take(context_path)
    ctx = json.loads(context_bytes)
    require(all(key in ctx for key in CONTEXT_KEYS), 'Missing actual release-context fields.')
    require(ctx['repository'] == REPOSITORY and ctx.get('edition', EDITION) == EDITION,
            'Unexpected repository or book edition.')
    require(ctx['pr_status'] == ctx['main_status'] == 'PASS',
            'Both actual PR/main CI PASS receipts are required; pending context refuses writes.')
    require(ctx['source_count'] == SOURCE_COUNT and ctx['source_set_sha256'] == SOURCE_SET,
            'Remote acceptance must identify the exact actual 98-source unified input.')
    for key in ('source_commit', 'source_tree', 'tested_main_commit', 'tested_main_tree'):
        require(isinstance(ctx[key], str) and re.fullmatch('[0-9a-f]{40}', ctx[key]),
                f'Missing full actual commit/tree SHA: {key}')
    for key in ('pull_request', 'pr_run', 'main_run'):
        require(type(ctx[key]) is int and ctx[key] > 0, f'Missing actual numeric identity: {key}')
    require(ctx['pr_run'] != ctx['main_run'], 'PR/main must be independent workflow runs.')
    now = ctx.get('recorded_at_utc')
    require(isinstance(now, str), 'recorded_at_utc must come from the actual release context.')
    when = datetime.fromisoformat(now.replace('Z', '+00:00'))
    require(when.utcoffset() is not None and when.utcoffset().total_seconds() == 0,
            'Actual metadata timestamp must include UTC.')

    integration = read(INTEGRATION)
    require(integration.get('schema') == 'ueot_qm.integration_verification.v31.9'
            and integration.get('status') == 'MAIN_VERIFIED', 'Root integration is not MAIN_VERIFIED.')
    for key in CONTEXT_KEYS:
        require(integration.get(key) == ctx[key], f'Actual context/integration mismatch: {key}')
    reconciliation_flags = ('tested_inputs_identical', 'core_imports_identical',
                            'canonical_book_identical', 'published_paths_identical')
    for key in reconciliation_flags:
        require(integration.get(key) is True, f'Root final-byte reconciliation missing: {key}')
    tree_equal = ctx['source_tree'] == ctx['tested_main_tree']
    require(integration.get('source_and_main_tree_identical') is tree_equal,
            'Whole-tree equality must reflect the actual tree identities.')
    require(integration.get('original_unified_raw_set_equals_publication_raw_set') is True,
            'This release has one exact raw source set, with no comment bridge.')
    require(integration.get('full_scientific_theory_verified') is False
            and integration.get('empirical_confirmation_added') is False,
            'Remote acceptance must retain finite formal scope and no new experiment.')

    local = read(LOCAL_GATE)
    require(local.get('schema') == 'ueot_physics.local_release.v3.5'
            and local.get('status') == LOCAL_STATE and local.get('remote_status') == 'PENDING'
            and local.get('remote_verified') is False, 'Immutable local book checkpoint differs.')
    for key in ('proof_gate', 'structure_gate', 'pdf_build_gate', 'visual_gate', 'dossier_gate'):
        require(local.get(key) == 'PASS', f'Actual local book gate missing: {key}')
    for key in ('canonical', 'pdf', 'latex', 'dossier', 'source_manifest', 'readme',
                'root_actual_local_acceptance', 'actual_source_manifest', 'structure', 'pdf_build',
                'visual_receipt', 'render_receipt', 'dossier_review', 'prior_preservation',
                'legacy_source_reuse', 'root_scientific_review', 'generator', 'context'):
        check_bound(local[key], 'immutable local book gate ' + key)
    if local.get('collaborator_visual_receipt') is not None:
        check_bound(local['collaborator_visual_receipt'], 'actual scoped collaborator visual receipt')
    counts = local['counts']
    require(local['source_count'] == SOURCE_COUNT and local['source_set_sha256'] == SOURCE_SET
            and counts['chapters'] == 66 and counts['appendices'] == 6
            and counts['numbered_results'] == 161 and counts['inherited'] == 147
            and counts['new'] == 14 and type(counts['pages']) is int and counts['pages'] > 0,
            'Current edition structure or exact source identity differs.')
    pages = counts['pages']
    canonical_hashes = {}
    for key, (name, root_key) in CANONICAL_KEYS.items():
        require(local[key] == record(BOOK + name), f'Current canonical binding changed: {name}')
        canonical_hashes[name] = local[key]['sha256']
        require(integration.get('book', {}).get(root_key) == canonical_hashes[name],
                f'Root remote canonical binding differs: {root_key}')
    for key, value in {'edition': EDITION, 'pages': pages, 'chapters': 66,
                       'appendices': 6, 'numbered_results': 161}.items():
        require(integration.get('book', {}).get(key) == value, f'Root remote book count differs: {key}')

    unified_identity = local['unified_source_identity']
    publication_identity = local['publication_source_identity']
    require(unified_identity['source_count'] == publication_identity['source_count'] == SOURCE_COUNT
            and unified_identity['source_set_sha256'] == publication_identity['source_set_sha256'] == SOURCE_SET
            and publication_identity['relation_to_unified_source'] == 'EXACT_SAME_BYTES',
            'Local and publication identity must bind identical raw input bytes.')
    for key in ('manifest', 'verification', 'source_count', 'source_set_sha256', 'actual_build_jobs'):
        require(integration['actual_unified_source_identity'].get(key) == unified_identity[key],
                f'Root relabelled the actual local unified identity: {key}')
    for key in ('manifest', 'source_count', 'source_set_sha256'):
        require(integration['publication_source_identity'].get(key) == publication_identity[key],
                f'Root publication source identity differs: {key}')
    require(unified_identity['manifest'] == publication_identity['manifest'] == local['actual_source_manifest']
            and unified_identity['verification'] == local['root_actual_local_acceptance'],
            'Book local and publication manifests are different objects.')
    proof = json.loads(check_bound(unified_identity['verification'], 'actual local unified proof'))
    manifest = json.loads(check_bound(unified_identity['manifest'], 'actual local unified manifest'))
    require(proof.get('schema') == 'ueot_qm.lean_verification.v31.9' and proof.get('status') == 'PASS'
            and proof['source_count'] == SOURCE_COUNT and proof['source_set_sha256'] == SOURCE_SET
            and proof['actual_build_exit_code'] == proof['actual_axiom_audit_exit_code'] == 0
            and proof['actual_build_jobs'] == unified_identity['actual_build_jobs']
            and proof['generic_modules'] == 41 and proof['explicit_theorems'] == 854
            and proof['actual_named_axiom_reports'] == 1123
            and proof['full_scientific_theory_verified'] is False, 'Actual unified proof identity differs.')
    require(manifest.get('schema') == 'ueot_qm.source_manifest.v2'
            and manifest['source_count'] == SOURCE_COUNT
            and source_set(manifest['sources']) == manifest['source_set_sha256'] == SOURCE_SET,
            'Source aggregate is not derived from the actual accepted input rows.')
    sources = rows_by_path(manifest['sources'], SOURCE_COUNT, 'accepted scientific sources')
    require(list(sources) == sorted(sources, key=lambda name: Path(name).parts),
            'Source paths must use the actual recorder Path-component order.')
    for relative, bound in sources.items():
        check_bound(bound, 'current accepted source ' + relative)
    freeze = read(KERNEL + 'source-freeze-reconciliation.json')
    require(freeze.get('status') == 'PASS' and freeze['source_count'] == SOURCE_COUNT
            and freeze['source_set_sha256'] == SOURCE_SET
            and freeze['all_source_bytes_unchanged_during_acceptance'] is True,
            'Actual source freeze is absent or changed.')

    # Preserve local execution history independently of later remote executions.
    finite = read(KERNEL + 'finite-witnesses/verification.json')
    new_finite = read(KERNEL + 'new-finite-witnesses/verification.json')
    reuse = read(KERNEL + 'legacy-reuse.json')
    execution = local['execution_partition']
    require(unified_identity['execution_partition'] == execution
            and execution['mode'] == finite['verification_mode'] == 'ACTUAL_NEW_AND_BOUND_HISTORICAL_REUSE'
            and finite['status'] == new_finite['status'] == reuse['status'] == 'PASS'
            and finite['actually_executed_witness_count'] == 3 and finite['reused_witness_count'] == 10
            and finite['witness_audit_count'] == 416 and len(finite['checks']) == 13,
            'Local witness execution/history partition differs.')
    for key, value in {'file_count': 3, 'theorems': 105, 'definitions': 39,
                       'abbreviations': 3, 'audits': 147, 'actually_executed_this_acceptance': True}.items():
        require(execution['new_witnesses'].get(key) == value, f'Actual new witness count differs: {key}')
    require(execution['new_witnesses']['receipt'] == record(KERNEL + 'new-finite-witnesses/verification.json')
            and new_finite['witness_audit_count'] == 147 and len(new_finite['checks']) == 3
            and new_finite['all_inputs_unchanged'] is True, 'Actual new witness receipt differs.')
    for key, value in {'file_count': 10, 'theorems': 193, 'definitions': 71,
                       'abbreviations': 5, 'audits': 269, 'actually_executed_this_acceptance': False}.items():
        require(execution['old_witnesses'].get(key) == value, f'Historical witness scope differs: {key}')
    for key in ('old_witnesses', 'exact_certificates', 'runtime', 'demos'):
        require(execution[key]['actually_executed_this_acceptance'] is False,
                f'Historical gate relabelled as a new local execution: {key}')
    require(reuse['unchanged_previous_source_count'] == 86 and reuse['core_source_count'] == 56
            and len(reuse['dependencies']) == 9 and reuse['witness_count'] == 10
            and reuse['new_legacy_lean_compiler_invocations'] == reuse['new_legacy_runtime_test_invocations'] == 0
            and proof['runtime_execution_mode'] == proof['demo_execution_mode'] == 'REUSED_UNCHANGED_SCIENTIFIC_INPUTS'
            and proof['instance_verification']['new_local_compiler_invocations'] == 0,
            'Immutable legacy reuse mode differs.')
    require(execution['reuse_review'] == record(KERNEL + 'legacy-reuse.json'),
            'Local book reuse binding differs.')

    # Read the real CI verifier receipts and their saved inputs. Do not reparse
    # logs or claim that detailed witness/demo artifact ZIP data were downloaded.
    ci_paths, ci = {}, {}
    ci_counts = {'generic_modules': 41, 'explicit_theorems': 854, 'generic_axiom_reports': 1123,
                 'printed_source_hash_count': 45, 'finite_witness_total_audits': 416,
                 'finite_witness_file_count': 13, 'finite_witness_theorem_count': 298,
                 'finite_witness_def_count': 110, 'finite_witness_abbrev_count': 8,
                 'runtime_test_methods': 100, 'demos': 4, 'positive_instances': 6,
                 'positive_instance_axiom_targets': 42, 'corrupted_true_rejections': 4,
                 'same_data_false_proofs': 4, 'false_witness_axiom_targets': 4,
                 'local_accepted_source_count': SOURCE_COUNT, 'publication_source_count': SOURCE_COUNT}
    for kind, commit, event in (('pr', ctx['source_commit'], 'pull_request'),
                                ('main', ctx['tested_main_commit'], 'push')):
        relative = KERNEL + f'ci/{kind}-{ctx[kind + "_run"]}-verification.json'
        require(integration.get(kind + '_verification') == relative,
                f'Root {kind} CI receipt path differs from its actual run.')
        receipt = read(relative)
        require(receipt.get('schema') == 'ueot_qm.actual_ci_verification.v31.9'
                and receipt.get('status') == 'PASS' and receipt.get('verification_complete') is True
                and receipt.get('repository') == REPOSITORY and receipt.get('event') == kind
                and receipt.get('workflow_event') == event and receipt.get('source_commit') == commit
                and receipt.get('run_id') == ctx[kind + '_run'], f'Wrong or incomplete actual {kind} CI.')
        for key in ('jobs_all_success', 'recorded_checkout_matches_expected',
                    'generic_axiom_names_and_foundation_equal_local', 'all_explicit_theorem_names_audited',
                    'printed_source_hashes_equal_local', 'printed_source_hashes_equal_publication_manifest',
                    'runtime_method_names_equal_local', 'input_files_unchanged_during_verification',
                    'local_accepted_sources_unchanged_during_verification',
                    'publication_sources_unchanged_during_verification',
                    'publication_matches_actual_local_source_bytes', 'local_witness_reference_validated',
                    'dependency_lock_gate_success', 'archive_steps_success'):
            require(receipt.get(key) is True, f'Actual {kind} verification incomplete: {key}')
        for key, value in ci_counts.items():
            require(receipt.get(key) == value, f'Actual {kind} count differs: {key}')
        require(receipt.get('allowed_foundation') == FOUNDATION
                and receipt.get('local_accepted_source_set_sha256') == SOURCE_SET
                and receipt.get('publication_source_set_sha256') == SOURCE_SET
                and receipt.get('local_reference_execution_mode') == 'THREE_NEW_WITNESSES_AND_BOUND_TEN_LEGACY_GATES'
                and receipt.get('remote_required_execution_mode') == 'ALL_CURRENT_WORKFLOW_GATES_MUST_ACTUALLY_EXECUTE',
                f'Actual {kind} foundation, source identity or execution separation differs.')
        require(receipt.get('full_scientific_theory_verified') is False
                and receipt.get('artifact_zip_downloaded') is False
                and receipt.get('artifact_zip_contents_verified') is False,
                f'Unexpected {kind} artifact/scope expansion requires an actual evidence review.')
        require(type(receipt.get('build_jobs')) is int and receipt['build_jobs'] > 0,
                f'Actual {kind} build-job count is missing.')
        for name, bound in receipt['input_files'].items():
            check_bound(bound, kind + ' CI verifier saved input ' + name)
        ci_paths[kind], ci[kind] = relative, receipt

    closure = read(KERNEL + 'core-import-closure.json')
    require(closure.get('transitive_module_count') == len(closure['modules']) == 56
            and closure.get('core_sources_modified') is False, 'Actual recursive Core closure differs.')
    core_rows = rows_by_path(closure['modules'], 56, 'actual recursive Core sources')
    for relative, row in core_rows.items():
        require(sha(take(path(relative))) == row['source_sha256'], f'Core source changed: {relative}')

    inventory_bound = integration['publication_inventory']
    inventory_bytes = check_bound(inventory_bound, 'actual frozen publication inventory')
    inventory = json.loads(inventory_bytes)
    published_count = inventory['file_count']
    require(inventory.get('schema') == 'ueot_qm.publication_inventory.v31.9'
            and inventory['source_count'] == SOURCE_COUNT and inventory['source_set_sha256'] == SOURCE_SET
            and type(published_count) is int and published_count > 0
            and integration['published_path_count'] == published_count,
            'Frozen publication inventory differs from the actual integration.')
    if 'original_unified_source_set_sha256' in inventory:
        require(inventory['original_unified_source_set_sha256'] == SOURCE_SET,
                'Frozen publication inventory invents a second local raw source set.')
    inventory_files = rows_by_path(inventory['files'], published_count, 'frozen publication inventory')
    if ctx.get('publication_inventory_path'):
        p = Path(ctx['publication_inventory_path'])
        require(take(p if p.is_absolute() else context_path.parent / p) == inventory_bytes,
                'Root integration and context identify different publication inventories.')
    checkpoint = integration['local_metadata_checkpoint']
    for key in ('source_commit', 'source_tree', 'tested_main_commit', 'tested_main_tree'):
        require(checkpoint.get(key) == ctx[key], f'Actual local metadata checkpoint identity differs: {key}')
    checkpoint_files = rows_by_path(checkpoint['files'], 12, 'pre-remote current metadata checkpoint')
    require(set(TARGETS[:3]) <= set(checkpoint_files), 'Book metadata absent from source-commit checkpoint.')
    for relative, row in checkpoint_files.items():
        require(relative in inventory_files, f'Checkpoint path was not actually published: {relative}')
        for key in ('sha256', 'bytes', 'git_blob_sha', 'mode'):
            require(row[key] == inventory_files[relative][key], f'Checkpoint/inventory mismatch: {relative} {key}')

    # Independently compare saved actual recursive trees with all frozen payload
    # blobs, while permitting unrelated concurrent work to change the whole tree.
    readback_inputs = integration['actual_recursive_tree_readback_inputs']
    trees = {}
    for kind, commit, tree in (('source', ctx['source_commit'], ctx['source_tree']),
                               ('main', ctx['tested_main_commit'], ctx['tested_main_tree'])):
        data = json.loads(check_bound(readback_inputs[kind], 'actual ' + kind + ' recursive tree readback'))
        expected_url = f'https://api.github.com/repos/{REPOSITORY}/git/trees/{tree}'
        require(data.get('commit') == commit and data.get('sha') == tree and data.get('truncated') is False
                and data.get('url', expected_url) == expected_url,
                f'Saved {kind} tree is not a complete actual commit-bound readback.')
        trees[kind] = rows_by_path(data['tree'], len(data['tree']), kind + ' actual Git tree')
    for relative, row in inventory_files.items():
        for kind in ('source', 'main'):
            actual_row = trees[kind].get(relative, {})
            require(actual_row.get('type') == 'blob' and actual_row.get('sha') == row['git_blob_sha']
                    and actual_row.get('mode') == row['mode'] and actual_row.get('size') == row['bytes'],
                    f'Actual {kind} published blob differs: {relative}')
        # Root's nine current metadata files may already have been finalized.
        # Their original bytes are resolved at source_commit via checkpoint.
        if relative not in checkpoint_files:
            data = check_bound(row, 'unchanged actual published file ' + relative)
            require(blob_sha(data) == row['git_blob_sha'], f'Published Git blob mismatch: {relative}')
    reconcile_ids = {}
    groups = {'tested_sources': set(sources), 'core_imports': set(core_rows),
              'canonical_book': {BOOK + name for name in canonical_hashes}}
    for kind in ('source', 'main'):
        relative = integration[kind + '_reconciliation']
        require(relative == KERNEL + f'remote-{kind}-reconciliation.json',
                f'Unexpected {kind} reconciliation path.')
        reconciliation = read(relative)
        require(reconciliation.get('schema') == f'ueot_qm.remote_{kind}_reconciliation.v31.9'
                and reconciliation.get('status') == 'PASS', f'Actual {kind} reconciliation is missing.')
        for key in ('source_commit', 'source_tree', 'tested_main_commit', 'tested_main_tree',
                    'source_count', 'source_set_sha256'):
            require(reconciliation.get(key) == ctx[key], f'{kind} reconciliation identity differs: {key}')
        for key in reconciliation_flags:
            require(reconciliation.get(key) is True, f'{kind} reconciliation incomplete: {key}')
        require(reconciliation.get('published_path_count') == published_count
                and reconciliation.get('source_and_main_tree_identical') is tree_equal
                and reconciliation.get('actual_recursive_tree_readback_inputs') == readback_inputs
                and reconciliation.get('publication_inventory') == inventory_bound,
                f'{kind} reconciliation input bindings differ.')
        for label, names in groups.items():
            actual_rows = rows_by_path(reconciliation[label], len(names), kind + ' ' + label)
            require(set(actual_rows) == names, f'{kind} {label} path population differs.')
            for name, row in actual_rows.items():
                data = check_bound(row, kind + ' reconciled ' + name)
                require(row['source_and_main_match'] is True and blob_sha(data) == row['git_blob_sha'],
                        f'{kind} recorded source/main blob match is invalid: {name}')
                for tree in trees.values():
                    require(tree[name]['type'] == 'blob' and tree[name]['sha'] == row['git_blob_sha']
                            and tree[name]['size'] == len(data), f'Actual reconciled tree blob differs: {name}')
        reconcile_ids[kind] = record(relative)

    visual = json.loads(check_bound(local['visual_receipt'], 'actual preserved root visual scope'))
    require(visual.get('status') == 'PASS' and visual.get('page_count') == pages
            and visual.get('pdf_sha256') == canonical_hashes['UEOT_PHYSICS.pdf']
            and visual.get('all_page_overview', {}).get('actually_viewed') is True,
            'Actual completed visual review does not bind this final PDF.')
    actual_pages = visual['root_full_pages_actually_viewed']
    require(isinstance(actual_pages, list) and actual_pages
            and all(type(x) is int and 1 <= x <= pages for x in actual_pages)
            and len(actual_pages) == len(set(actual_pages)), 'Invalid actual full-page review scope.')

    originals = {relative: take(path(relative)) if path(relative).is_file() else None for relative in TARGETS}
    require(originals[REMOTE] is None, 'Remote receipt already exists; reconcile without overwriting history.')
    previous = {}
    for relative, gate_key, previous_key in zip(TARGETS[:3], ('source_manifest', 'dossier', 'readme'),
                                                ('source_manifest', 'chapter_dossier', 'readme')):
        data = originals[relative]
        require(data is not None and local[gate_key] == record(relative),
                f'Current accepted local book metadata changed: {relative}')
        for bound in (inventory_files[relative], checkpoint_files[relative]):
            require(bound['sha256'] == sha(data) and bound['bytes'] == len(data)
                    and bound['git_blob_sha'] == blob_sha(data),
                    f'Source-commit metadata binding differs: {relative}')
        previous[previous_key] = {**record(relative), 'git_blob_sha': blob_sha(data),
                                 'source_commit': ctx['source_commit'], 'source_tree': ctx['source_tree'],
                                 'historical_resolution': 'Read this path at source_commit; later metadata has a separate byte identity.'}

def main():
    global WRITES
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root', type=Path, required=True)
    parser.add_argument('--context', type=Path, required=True)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument('--apply', action='store_true')
    mode.add_argument('--plan', action='store_true')
    mode.add_argument('--dry-run', action='store_true')
    args = parser.parse_args()
    if args.plan:
        print(json.dumps({'status': 'PLAN_ONLY_NOT_EXECUTED', 'targets': list(TARGETS),
                          'context_required': list(CONTEXT_KEYS) + ['recorded_at_utc'],
                          'requires': 'Actual independent PR/main CI PASS, root MAIN_VERIFIED integration, actual frozen-path reconciliation, source-commit 12-file checkpoint, final immutable 3.5 local book gates.',
                          'source_count': SOURCE_COUNT, 'source_set_sha256': SOURCE_SET,
                          'default_read_only': True, 'no_scientific_reexecution': True,
                          'completed_repository_writes': 0}, ensure_ascii=False, indent=2))
        return
    prepared, originals, protected, summary = prepare(args.root, args.context)
    if args.apply:
        root = args.root.resolve()
        for p, digest in protected.items():
            require(p.is_file() and sha(p.read_bytes()) == digest, f'Protected input changed before apply: {p}')
        for relative, data in originals.items():
            p = root / relative
            require(p.resolve().is_relative_to(root / 'physics/qm/book'), f'Unsafe write target: {relative}')
            require((p.read_bytes() if p.is_file() else None) == data, f'Metadata changed concurrently: {relative}')
        # Only four atomic replacements. A failure reports actual completed
        # writes and never overwrites historical evidence to simulate success.
        for relative in TARGETS:
            target, temporary = root / relative, None
            try:
                with tempfile.NamedTemporaryFile(prefix='.' + target.name + '.qm319-',
                                                 dir=target.parent, delete=False) as stream:
                    temporary = Path(stream.name)
                    stream.write(prepared[relative])
                    stream.flush()
                    os.fsync(stream.fileno())
                temporary.replace(target)
                temporary = None
                WRITES += 1
            finally:
                if temporary is not None:
                    temporary.unlink(missing_ok=True)
        for relative in TARGETS:
            require((root / relative).read_bytes() == prepared[relative], f'Metadata readback failed: {relative}')
        for p, digest in protected.items():
            require(p.is_file() and sha(p.read_bytes()) == digest, f'Protected bytes changed during apply: {p}')
        summary['status'] = 'MAIN_VERIFIED_BOOK_METADATA_APPLIED'
        summary['completed_repository_writes'] = WRITES
    print(json.dumps(summary, ensure_ascii=False, indent=2))


if __name__ == '__main__':

        main()
    except (OSError, ValueError, KeyError, TypeError, UnicodeError) as exc:
        print(json.dumps({'status': 'REFUSED_OR_FAILED', 'error': str(exc),
                          'completed_repository_writes': WRITES}, ensure_ascii=False), file=sys.stderr)
        sys.exit(2)
