def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def names(namespace, *xs):
    return ['UEOTQM.' + namespace + '.' + x for x in xs]

def decls(p):
    s = p.read_text()
    ns = re.search(r'^namespace (\S+)', s, re.M).group(1)
    return [{'kind': m[0], 'name': ns + '.' + m[1]} for m in
            re.findall(r'^(?:@\[[^\]]+\]\s*)?(theorem|def|abbrev)\s+(\S+)', s, re.M)]

def parse(p):
    raw = p.read_text()
    assert 'error:' not in raw and 'sorryAx' not in raw and 'declaration uses sorry' not in raw
    result = []
    for m in re.finditer(r"'([^']+)' (?:depends on axioms:\s*\[([^\]]*)\]|does not depend on any axioms)", raw, re.S):
        axs = [] if m[2] is None else [a.strip() for a in m[2].split(',') if a.strip()]
        assert set(axs) <= ALLOWED
        result.append({'name': m[1], 'axioms': axs})
    assert len(result) == len({x['name'] for x in result})
    return result
