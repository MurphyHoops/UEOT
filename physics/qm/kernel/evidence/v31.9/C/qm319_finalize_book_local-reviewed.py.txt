def require(test, message):
    if not test:
        raise ValueError(message)

def digest(raw): return sha256(raw).hexdigest()
def json_bytes(value): return (json.dumps(value, ensure_ascii=False, indent=2)+'\n').encode()
def identity(path, raw): return {'path':str(path), 'sha256':digest(raw), 'bytes':len(raw)}

def parse_claims(text):
    lines=text.split('<a id="appendix-a"></a>',1)[0].splitlines()
    chapter=None; rows=[]
    for i,line in enumerate(lines):
        heading=re.fullmatch(r'# (\d+)\. .+',line)
        if heading: chapter=int(heading[1])
        anchor=re.fullmatch(r'<a id="([a-z0-9-]+)"></a>',line)
        if not anchor or anchor[1].startswith('ch-'): continue
        result_id=anchor[1].upper()
        following=next((x for x in lines[i+1:] if x.strip()),'')
        title=re.match(r'\*\*(.+?)\*\*',following)
        require(title and re.match(re.escape(result_id)+r'(?=$|[^A-Za-z0-9_-])',title[1]),'Invalid numbered anchor: '+result_id)
        rows.append({'id':result_id,'chapter':chapter,'title':title[1]})
    return rows

def body_slices(text, chapters):
    marks=list(re.finditer(r'^<a id="(ch-\d+|appendix-[a-f])"></a>$',text,re.M))
    expected=['ch-'+str(n) for n in range(chapters)]+['appendix-'+a for a in 'abcdef']
    require([m[1] for m in marks]==expected,'Unexpected ordered chapter/appendix anchors')
    return {m[1]:text[m.start():marks[i+1].start() if i+1<len(marks) else len(text)] for i,m in enumerate(marks)}

def embedded_programs(text):
    region=text.split('# 附录 D.',1)[1].split('<a id="appendix-e">',1)[0]
    return re.findall(r'(?ms)^~~~python\n(.*?)^~~~\s*$',region)
