ALLOWED = {'propext', 'Classical.choice', 'Quot.sound'}

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def declarations(p):
    s = p.read_text()
    ns = re.search(r'^namespace (\S+)', s, re.M).group(1)
    return [{'kind': m[0], 'name': ns + '.' + m[1]} for m in
            re.findall(r'^(?:@\[[^\]]+\]\s*)?(theorem|def|abbrev)\s+(\S+)', s, re.M)]

def audits(p):
    s = p.read_text()
    result = []
    for m in re.finditer(r"'([^']+)' (?:depends on axioms:\s*\[([^\]]*)\]|does not depend on any axioms)", s, re.S):
        axs = [] if m[2] is None else [x.strip() for x in m[2].split(',') if x.strip()]
        assert set(axs) <= ALLOWED, (m[1], axs)
        result.append({'name': m[1], 'axioms': axs})
    assert len({x['name'] for x in result}) == len(result)
    assert 'error:' not in s and 'sorryAx' not in s and 'declaration uses sorry' not in s
    return result

        'QuantumTraceDistanceExamples-compile-002']
bindings = []
decls = []
for tag in tags:
    rp = E / 'A' / (tag + '.json')
    r = json.loads(rp.read_text())
    p, op = ROOT / r['source'], ROOT / r['olean_path']
    sp, lp = ROOT / r['source_snapshot'], ROOT / r['log']
    assert r['exit_code'] == 0 and r['source_unchanged'] is True
    assert sha(p) == r['source_sha256'] == sha(sp)
    assert sha(op) == r['olean_sha256']
    assert p.read_bytes() == sp.read_bytes()
    assert not re.search(r'\b(sorry|admit|axiom)\b', p.read_text().split('/- Every concrete')[0])
    assert 'error:' not in lp.read_text() and 'declaration uses sorry' not in lp.read_text()
    ds = declarations(p)
    decls.append(ds)
    bindings.append({'source': r['source'], 'source_sha256': sha(p),
                     'source_snapshot': r['source_snapshot'], 'source_snapshot_sha256': sha(sp),
                     'olean': r['olean_path'], 'olean_sha256': sha(op),
                     'receipt': str(rp.relative_to(ROOT)), 'receipt_sha256': sha(rp),
                     'log': r['log'], 'log_sha256': sha(lp), 'exit_code': r['exit_code'],
                     'declaration_census': dict(collections.Counter(x['kind'] for x in ds))})
