bound = {}


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def blob(raw):
    return hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()


def read_bytes(path):
    p = Path(path)
    if not p.is_absolute():
        p = ROOT / p
    raw = p.read_bytes()
    key = p.relative_to(ROOT).as_posix() if p.is_relative_to(ROOT) else str(p)
    item = {'sha256': sha(raw), 'bytes': len(raw)}
    assert key not in bound or bound[key] == item, 'Input drift during independent review: ' + key
    bound[key] = item
    return raw


def read_json(path):
    return json.loads(read_bytes(path))


def check_identity(item):
    raw = read_bytes(item['path'])
    assert sha(raw) == item['sha256'], item['path']
    assert len(raw) == item.get('bytes', item.get('size')), item['path']
    return raw


def parse_axioms(raw):
    text = raw.decode()
    pattern = r"'([^'\n]+)' (?:depends on axioms:\s*\[([^\]]*)\]|does not depend on any axioms)"
    matches = list(re.finditer(pattern, text))
    names = [m[1] for m in matches]
    assert len(names) == len(set(names)), 'Duplicate actual axiom output'
    actual = {m[1]: sorted(x.strip() for x in (m[2] or '').split(',') if x.strip()) for m in matches}
    assert all(set(v) <= ALLOWED for v in actual.values()), 'Unreviewed axiom'
    return names, actual

    paths.add(ROOT / '.github/workflows/ueot-qm.yml')
    return sorted(paths)
