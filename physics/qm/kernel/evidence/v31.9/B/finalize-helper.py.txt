bind = lambda p: {'path': str(p.relative_to(ROOT)) if p.is_relative_to(ROOT) else str(p), 'bytes': p.stat().st_size, 'sha256': sha(p)}

checked = []
for path, name in zip(files, passes):

    d = json.loads(p.read_text())
    assert d['exit_code'] == 0 and d['status'] == 'PASS' and d['source_unchanged_during_execution']
    assert d['source_sha256_before'][str(path.relative_to(ROOT))] == sha(path)
    assert d['source_sha256_after'][str(path.relative_to(ROOT))] == sha(path)
    log = ROOT / d['log']
    assert sha(log) == d['log_sha256']
    item = {'source': bind(path), 'execution': bind(p), 'log': bind(log)}
    if not path.name.endswith('Examples.lean'):
        o = ROOT / ('formalization/ueot-qm/.lake/build/lib/lean/UEOTQM/' + path.stem + '.olean')
        assert sha(o) == d['olean_sha256']
        item['olean'] = bind(o)
    checked.append(item)
pub = json.loads((E / 'public-audit-001.json').read_text())
assert pub['status'] == 'PASS' and pub['exit_code'] == 0 and pub['inputs_unchanged']
assert pub['actual_count'] == pub['requested_count'] == 29

assert set(actual) == set(requested) and len(actual) == 34
assert all(x == 1 for x in Counter(requested).values()) and all(x == 1 for x in Counter(n for n,a in reports).values())
assert all(set(a) <= allowed for a in actual.values())
declarations = []

    declarations.append({'source': bind(p), 'declarations': ds, 'counts': dict(Counter(d['kind'] for d in ds))})
assert sum(d['counts'].get('theorem',0) for d in declarations[:2]) == 23
assert declarations[-1]['counts'] == {'def':9, 'theorem':25}

       'scientific_scope':['Exact purification existence iff the actual density matrix rank fits the finite auxiliary cardinality','Attained purification on Fin rank and impossibility on smaller or empty auxiliary carriers','Actual direct auxiliary isometry between arbitrary full convex-pure states under equal marginals and the stated cardinal direction'],

       'book_chapter_draft':bind(H/'B_v319_book_chapter.md'),'new_claim_draft':bind(H/'B_v319_claims.json'),'book_result_ids':['P-PDM-01','P-PDM-02','P-PDM-03','P-PDM-04','N-PDM-01'],
