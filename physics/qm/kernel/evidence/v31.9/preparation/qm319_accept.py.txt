def process_tree(pid):
    """Read current descendants, including verifier children with their own sessions."""
    table = {}
    for path in Path('/proc').iterdir():
        if not path.name.isdigit():
            continue
        try:
            fields = (path / 'stat').read_text().rsplit(')', 1)[1].split()
            table[int(path.name)] = {'ppid':int(fields[1]), 'pgrp':int(fields[2]), 'start':fields[19]}
        except (FileNotFoundError, ProcessLookupError, PermissionError):
            continue
    selected = {pid}
    while True:
        expanded = selected | {p for p, row in table.items() if row['ppid'] in selected}
        if expanded == selected:
            break
        selected = expanded
    return {p:table[p] for p in selected if p in table}

def stop_process_tree(process):
    snapshot = process_tree(process.pid)
    groups = {row['pgrp'] for row in snapshot.values()} | {process.pid}
    groups.discard(os.getpgrp())
    for group in groups:
        try: os.killpg(group, signal.SIGTERM)
        except ProcessLookupError: pass
    try: process.wait(timeout=5)
    except subprocess.TimeoutExpired: pass
    # Recheck known process start times before killing any surviving group.
    # This also covers grandchildren launched in separate verifier sessions.
    surviving_groups = set()
    for pid, row in snapshot.items():
        try:
            fields = (Path('/proc') / str(pid) / 'stat').read_text().rsplit(')', 1)[1].split()
            if fields[19] == row['start']:
                surviving_groups.add(int(fields[2]))
        except (FileNotFoundError, ProcessLookupError, PermissionError):
            pass
    surviving_groups.discard(os.getpgrp())
    for group in surviving_groups:
        try: os.killpg(group, signal.SIGKILL)
        except ProcessLookupError: pass
    process.wait()
    return {'termination_requested_groups':sorted(groups), 'killed_surviving_groups':sorted(surviving_groups),
            'parent_reaped':True}

def run(name, command, cwd=P, timeout=1800, output=None):
    log = Path(output) if output is not None else E / (name + '.log')
    start = now(); clock = time.monotonic()
    print('RUN ' + name, flush=True)
    timed_out = False; launch_error = None; exit_code = None; cleanup = None
    with log.open('wb') as stream:
        try:
            process = subprocess.Popen(command, cwd=cwd, stdout=stream, stderr=subprocess.STDOUT,
                                       start_new_session=True)
            try:
                exit_code = process.wait(timeout=timeout)
            except BaseException:
                cleanup = stop_process_tree(process)
                raise
        except subprocess.TimeoutExpired as exc:
            timed_out = True
            stream.write(('\nACCEPTANCE_TIMEOUT: '+repr(exc)+'\n').encode())
        except OSError as exc:
            launch_error = repr(exc)
            stream.write(('\nACCEPTANCE_LAUNCH_ERROR: '+launch_error+'\n').encode())
    receipt = {'name':name, 'command':command, 'working_directory':str(cwd),
               'started_at_utc':start, 'completed_at_utc':now(), 'elapsed_seconds':round(time.monotonic()-clock,3),
               'exit_code':exit_code, 'timed_out':timed_out, 'launch_error':launch_error,
               'process_tree_cleanup':cleanup,
               'log_path':str(log), 'log_sha256':sha(log)}
    executions.append(receipt)
    write(E / 'execution-journal.json', {'schema':'ueot_qm.execution_journal.v31.9','executions':executions})
    print(json.dumps(receipt, ensure_ascii=False), flush=True)
    if exit_code != 0:
        print(log.read_text(errors='replace')[-12000:], flush=True)
        raise RuntimeError('Execution failed: ' + name)
    return receipt
