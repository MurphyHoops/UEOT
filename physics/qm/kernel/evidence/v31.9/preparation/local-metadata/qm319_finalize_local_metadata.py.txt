class Refused(RuntimeError):
    pass


def require(test, reason):
    if not test:
        raise Refused(reason)


def now():
    return datetime.now(timezone.utc).isoformat()


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def encoded(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode()


def read(path):
    return json.loads(Path(path).read_bytes())


def safe(root, rel):
    require(isinstance(rel, str) and rel and not Path(rel).is_absolute(), "Expected repository-relative path")
    p = (root / rel).resolve()
    require(p.is_relative_to(root) and p != root, "Path escapes the repository: " + rel)
    return p


def ident(root, path):
    p = Path(path).resolve()
    raw = p.read_bytes()
    return {"path": p.relative_to(root).as_posix() if p.is_relative_to(root) else str(p),
            "sha256": digest(raw), "bytes": len(raw)}


def check_ident(root, item, *, allow_external=False):
    require(isinstance(item, dict) and {"path", "sha256", "bytes"} <= set(item), "Incomplete file identity")
    if Path(item["path"]).is_absolute():
        require(allow_external, "Unexpected external identity: " + item["path"])
        p = Path(item["path"]).resolve()
        require(p.is_relative_to(root.parent / "handoff"), "External identity is outside task handoff")
    else:
        p = safe(root, item["path"])
    require(p.is_file(), "Missing bound input: " + str(p))
    actual = ident(root, p)
    require(actual["sha256"] == item["sha256"] and actual["bytes"] == item["bytes"], "Stale bound input: " + str(p))
    return p


def blob(raw):
    return hashlib.sha1(b"blob " + str(len(raw)).encode() + b"\0" + raw).hexdigest()


def source_paths(root):
    package = root / "formalization/ueot-qm"
    paths = set((package / "UEOTQM").rglob("*.lean"))
    paths.update(package / name for name in ("UEOTQM.lean", "lakefile.lean", "lake-manifest.json", "lean-toolchain"))
    paths.update(p for p in (package / "scripts").rglob("*") if p.is_file() and p.suffix in {".py", ".lean", ".sh"})
    kernel = root / "physics/qm/kernel"
    paths.update(kernel.rglob("*.py"))
    paths.update(p for p in (kernel / "certificates").rglob("*") if p.is_file() and p.suffix in {".lean", ".json"})
    paths.add(root / ".github/workflows/ueot-qm.yml")
    # Match the actual collector's Path-component ordering, not lexical strings.
    return sorted(paths)


def current_sources(root):
    rows = [ident(root, p) for p in source_paths(root)]
    return rows, digest(json.dumps(rows, sort_keys=True, separators=(",", ":")).encode())


def original(ctx_dir, ctx, rel):
    item = ctx["originals"][rel]
    if not item["exists"]:
        return None
    p = ctx_dir / item["snapshot"]
    raw = p.read_bytes()
    require(digest(raw) == item["sha256"] and len(raw) == item["bytes"], "Changed original metadata snapshot: " + rel)
    return raw


def prepare_context(root, path):
    require(not path.exists(), "Context already exists; original snapshots must not be replaced")
    require(path.parent == root.parent / "handoff", "Keep task context in handoff")
    tree_path = path.parent / "qm319_baseline_remote_tree.json"
    tree = read(tree_path)
    require(tree["commit"] == BASE and tree["sha"] == BASE_TREE and not tree["truncated"], "Wrong actual baseline tree")
    by_path = {x["path"]: x["sha"] for x in tree["tree"] if x["type"] == "blob"}
    originals = {}
    snapshot_dir = path.parent / "qm319_metadata_originals"
    require(not snapshot_dir.exists(), "Original metadata snapshot directory already exists")
    for rel in OWNED:
        p = safe(root, rel)
        if not p.exists():
            originals[rel] = {"exists": False}
            continue
        raw = p.read_bytes()
        if rel in by_path:
            require(blob(raw) == by_path[rel], "Owned original changed from the actual baseline before snapshot: " + rel)
        q = snapshot_dir / rel
        q.parent.mkdir(parents=True, exist_ok=True)
        q.write_bytes(raw)
        originals[rel] = {"exists": True, "snapshot": q.relative_to(path.parent).as_posix(), "sha256": digest(raw), "bytes": len(raw)}
    old = safe(root, OLD_LEDGER)
    require(blob(old.read_bytes()) == by_path[OLD_LEDGER], "Historical ledger differs from actual baseline")
    ctx = {"schema": "ueot_qm.local_metadata_context.v31.9", "status": "PREPARED_NOT_APPLIED",
           "prepared_at_utc": now(), "repository": str(root), "baseline_commit": BASE, "baseline_tree": BASE_TREE,
           "baseline_tree_receipt": ident(root, tree_path), "source_count_required": SOURCE_COUNT,
           "owned_paths": list(OWNED), "originals": originals, "previous_ledger": ident(root, old),
           "proof_directory": PROOF_DIR, "book_evidence_directory": BOOK_DIR,
           "scientific_review": PROOF_DIR + "/root-scientific-review.json",
           "gate_rule": "Rendering and apply both require actual unified proof plus completed book gates. No planned result can authorize LOCAL_VERIFIED_REMOTE_PENDING."}
    path.write_bytes(encoded(ctx))
    return {"status": "CONTEXT_PREPARED_NOT_APPLIED", "context": ident(root, path), "repository_files_changed": 0}


def protected_inputs(root, baseline, core, source, book_files):
    protected = {}
    for rel, expected_blob in baseline.items():
        historical = (rel.startswith("physics/qm/kernel/evidence/") or rel.startswith("physics/qm/book/evidence/")
                      or rel.startswith("physics/qm/audits/") or rel.startswith("physics/qm/status/")
                      or (rel.startswith("physics/qm/UEOT_QM_") and rel.endswith(".md")))
        if historical and rel not in OWNED:
            p = safe(root, rel)
            require(p.is_file() and blob(p.read_bytes()) == expected_blob, "Historical file changed: " + rel)
            protected[rel] = ident(root, p)
    for row in source["sources"]:
        p = check_ident(root, row)
        protected[row["path"]] = ident(root, p)
    require(core["transitive_module_count"] == len(core["modules"]) == 56, "Wrong actual recursive Core closure")
    for row in core["modules"]:
        p = safe(root, row["path"])
        raw = p.read_bytes()
        require(digest(raw) == row["source_sha256"] and blob(raw) == baseline[row["path"]], "Frozen Core identity changed: " + row["module"])
        require(row["baseline_match"] is True, "Core baseline comparison did not pass")
        protected[row["path"]] = ident(root, p)
    for item in book_files:
        p = check_ident(root, item)
        protected[item["path"]] = ident(root, p)
    # Own no receipts: all existing current proof/book evidence must remain fixed.
    for folder in (PROOF_DIR, BOOK_DIR):
        for p in sorted((root / folder).rglob("*")):
            if p.is_file():
                protected[p.relative_to(root).as_posix()] = ident(root, p)
    require(not set(protected).intersection(OWNED), "An owned metadata path entered the scientific freeze")
    return protected


def normalize_binding(item, chapter):
    result = {"book_id": item["id"], "anchor": item["anchor"], "chapter": chapter,
              "title": item["title"], "premises": item["premises"], "quantified_scope": item["quantified_scope"],
              "claim": item.get("claim", item.get("actual_conclusion")), "main": item["main_declarations"],
              "support": item.get("support_declarations", []), "witnesses": item["witnesses"],
              "proof_outline": item["proof_outline"], "limits": item["limits"],
              "external_declarations": item.get("external_declarations", []),
              "evidence_kind": item.get("evidence_kind", "EXACT_FORMAL_NEGATIVE_CONTROL" if item["id"].startswith("N-") else "FORMAL_THEOREM")}
    for key in ("title", "premises", "quantified_scope", "claim", "main", "witnesses", "proof_outline", "limits"):
        require(bool(result[key]), "Incomplete reviewed result: " + result["book_id"] + "/" + key)
    require(result["anchor"] == result["book_id"].lower(), "Unstable numbered anchor")
    return result

def between(text, start, end):
    require(text.count(start) == 1 and text.count(end) == 1, "Ambiguous retained documentation section")
    a, b = text.index(start), text.index(end)
    require(a < b, "Reversed documentation section")
    return text[a:b]


def paragraph_replace(text, prefix, replacement):
    require(text.count(prefix) == 1, "Ambiguous paragraph: " + prefix)
    a = text.index(prefix)
    b = text.find("\n\n", a)
    b = len(text) if b < 0 else b
    return text[:a] + replacement + text[b:]

def validate_rendered(root, rendered, d, ledger):
    require(set(rendered) == set(OWNED), "Output ownership mismatch")
    for rel in OWNED[:3]:
        item = json.loads(rendered[rel])
        if rel == OWNED[0]:
            require(item["verification_state"] == STATE and item["integration"]["state"] == "REMOTE_PENDING", "CURRENT state mismatch")
            require(item["integration"]["source_identity"]["source_set_sha256"] == d["source_sha"], "CURRENT identity mismatch")
            require(all(item["integration"][k] is None for k in ("source_commit", "source_tree", "source_pr", "pr_run", "tested_main_commit", "main_run")), "Invented current remote identity")
        elif rel == OWNED[1]:
            require(item["checkpoint"]["state"] == STATE and item["checkpoint"]["book"] == "LOCAL_GATES_PASS", "Current ACTIVE checkpoint still in progress")
        else:
            require(item["verification"]["remote_status"] == "PENDING" and item["claims"][:70] == d["old_ledger"]["claims"], "Ledger current/historical scope mismatch")
    # Validate local Markdown links against both actual files and this exact output.
    for rel, raw in rendered.items():
        if not rel.endswith(".md"):
            continue
        text = raw.decode()
        for target in re.findall(r'\]\(([^\s)]+)\)', text):
            if target.startswith(("http:", "https:", "mailto:", "#")):
                continue
            target_path = target.split("#", 1)[0]
            if not target_path:
                continue
            path = (safe(root, rel).parent / target_path).resolve()
            require(path.is_relative_to(root), "Documentation link escapes repository")
            require(path.is_file() or path.is_dir() or path.relative_to(root).as_posix() in rendered, "Broken local navigation link: " + rel + " -> " + target)
    require(ledger["claims"][:70] == d["old_ledger"]["claims"], "Historical object rewrite")
