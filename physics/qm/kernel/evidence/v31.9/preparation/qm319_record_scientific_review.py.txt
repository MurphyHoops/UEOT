This is a light identity/claim reconciliation, not another proof execution.
Author claim records and historical focused receipts remain unmodified.
"""
from pathlib import Path
from datetime import datetime, timezone
import hashlib
import importlib.util
import json

import shutil

R = Path('/workspace/scratch/f11b4261e187/qm_v318_work')
H = Path(__file__).resolve().parent
E = R / 'physics/qm/kernel/evidence/v31.9'
B = R / 'physics/qm/book'
FOUNDATION = {'propext', 'Classical.choice', 'Quot.sound'}

def read(p):
    return json.loads(Path(p).read_bytes())

def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def identity(p):
    p = Path(p)
    try:
        name = str(p.relative_to(R))
    except ValueError:
        name = str(p)
    return {'path': name, 'sha256': sha(p), 'bytes': p.stat().st_size}

def write(p, d):
    p = Path(p)
    assert not p.exists(), 'Preserve previous actual review: ' + str(p)
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(d, ensure_ascii=False, indent=2) + '\n')

    for path, row in sources.items():
        current = identity(R / path)
        assert current == row, path

    spec = importlib.util.spec_from_file_location('qm319_parser', H / 'qm319_source_parser.py')
    parser = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(parser)
    locations = {}
    for module, rows in inventory['modules'].items():
        path = 'formalization/ueot-qm/UEOTQM/' + module + '.lean'
        for row in rows:
            if row['name'] in axioms['declarations']:
                locations[row['name']] = {**sources[path], 'line': row['line'], 'kind': row['kind'],
                    'execution_mode': 'EXECUTED_CURRENT_UNIFIED_PUBLIC_BUILD_AND_AUDIT',
                    'actual_named_audit_receipt': 'physics/qm/kernel/evidence/v31.9/proof-axioms.json'}

        name = check['name']
        path = 'formalization/ueot-qm/scripts/witnesses/' + name + '.lean'

        actual = check['audit']['actual']
        assert check['status'] == check['audit']['status'] == 'PASS' and check['exit_code'] == 0
        assert check['audit']['complete_unique'] and set(actual) == set(check['audit']['expected'])

        assert all(set(v) <= FOUNDATION for v in actual.values())
        assert sha(R / path) == check['source']['sha256'] == sources[path]['sha256']
        for row in rows:
            assert row['name'] not in locations and row['name'] not in witness_names
            witness_names.add(row['name'])
            locations[row['name']] = {**sources[path], 'line': row['line'], 'kind': row['kind'],
                'execution_mode': check['execution_mode'],
                'actual_named_audit_receipt': 'physics/qm/kernel/evidence/v31.9/finite-witnesses/verification.json'}

    library_text = library_path.read_text()

        pattern = r'^theorem ' + re.escape(name.split('.')[-1]) + r'\b'

            'name': name,
            'package': 'mathlib',

            names = list(dict.fromkeys(main + support + controls))
            assert names and set(names) <= all_names and set(names) <= locations.keys()
            assert not set(names) & set(ext)
            for key in ['premises', 'quantified_scope', 'proof_outline', 'limits']:
                assert row[key], (identifier, key)
            claim = row.get('claim', row.get('actual_conclusion'))
            assert claim
            normalized.append({
                'id': identifier, 'anchor': anchor, 'title': book_by_id[identifier]['title'],
                'author_title': row['title'], 'premises': row['premises'],
                'quantified_scope': row['quantified_scope'], 'claim': claim,
                'main_declarations': main, 'support_declarations': support, 'witnesses': controls,
                'proof_outline': row['proof_outline'], 'limits': row['limits'],

                'classification_adjustments': classifications,

                'new_experimental_data': False,

            bound_names.update(names)

        write(out, {'schema': 'ueot_qm.reviewed_chapter_bindings.v31.9', 'status': 'PASS_ACTUAL_LOCAL_BINDING',
            'chapter': chapter, 'author_claim_snapshot': identity(original),
            'canonical_chapter_input': identity(draft), 'canonical_source': identity(B / 'UEOT_PHYSICS.md'),
            'actual_unified_proof_receipt': identity(E / 'lean-verification.json'),
            'actual_public_audit_receipt': identity(E / 'proof-axioms.json'),
            'actual_witness_audit_receipt': identity(E / 'finite-witnesses/verification.json'),
            'source_set_sha256': proof['source_set_sha256'], 'results': normalized,
            'scope': 'Normalized current claim/evidence mapping. Original author/focused checkpoint statuses retain their historical scope in the raw snapshot.'})
        chapters.append({'chapter': chapter, 'chapter_bindings': identity(out),
            'all_bound_declaration_names_actually_audited': True,
            'body_integrated_exactly': True, 'result_count': len(normalized)})

            'bound_historical_files': 10, 'bound_historical_audits': 269,

            'legacy_reuse': identity(E / 'legacy-reuse.json')},

        'cross_reviews': reviews,
        'review_method': 'Root source/argument inspection during implementation, three cross-lane independent reviews, and current actual compiler/audit outputs reconciled to every new numbered claim.         ],
