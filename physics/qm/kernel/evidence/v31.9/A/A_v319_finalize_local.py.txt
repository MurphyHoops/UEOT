from pathlib import Path

def sha(path): return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def read(path): return json.loads(Path(path).read_bytes())
def write_new(path, data):
    assert not path.exists(), path
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
def rel(path): return Path(path).relative_to(R).as_posix()

spec = importlib.util.spec_from_file_location('a_witness_readback', P/'scripts/verify_finite_witnesses.py')
verifier = importlib.util.module_from_spec(spec)
spec.loader.exec_module(verifier)
allowed = {'propext', 'Classical.choice', 'Quot.sound'}
public_rows, public_bindings = [], []
for stem, attempt in [('QuantumTraceDistance', 3), ('BinaryDiscrimination', 1)]:
    path = P/'UEOTQM'/(stem+'.lean')
    receipt_path = E/f'{stem}-compile-{attempt:03d}.json'
    receipt = read(receipt_path)
    rows, diagnostics, instances = declarations(path.read_text())
    assert not diagnostics and not instances
    assert receipt['exit_code'] == 0 and receipt['source_unchanged']
    assert sha(path) == receipt['source_sha256'] == sha(R/receipt['source_snapshot'])
    assert sha(R/receipt['olean_path']) == receipt['olean_sha256']
    public_rows.extend(rows)
    public_bindings.append({'source':rel(path), 'source_sha256':sha(path),
      'source_bytes':path.stat().st_size, 'olean':receipt['olean_path'],
      'olean_sha256':receipt['olean_sha256'], 'compile_receipt':rel(receipt_path),
      'compile_receipt_sha256':sha(receipt_path), 'compile_log':receipt['log'],
      'compile_log_sha256':sha(R/receipt['log']), 'exit_code':0})

assert audit['source_and_olean_unchanged']

assert verifier.audit_result(E/'PublicAxioms-001.log', public_rows)['status'] == 'PASS'

    assert old['path'] == actual['source'] and old['sha256'] == actual['source_sha256']
    assert old['olean_sha256'] == actual['olean_sha256']
