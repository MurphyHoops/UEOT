def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha_bytes(data):
    return hashlib.sha256(data).hexdigest()


def sha(path):
    return sha_bytes(Path(path).read_bytes())


def read(path):
    return json.loads(Path(path).read_text())


def rel(path):
    return Path(path).relative_to(R).as_posix()


def encode(obj):
    return (json.dumps(obj, ensure_ascii=False, indent=2) + '\n').encode()


def file_record(path, prepared):
    path = Path(path)
    data = prepared[path] if path in prepared else path.read_bytes()
    return {'path': rel(path), 'sha256': sha_bytes(data), 'bytes': len(data)}


def collect_paths(obj):
    result = set()
    if isinstance(obj, dict):
        for key, value in obj.items():
            if key in ['path', 'manifest_path', 'catalog_path', 'source_manifest_path',
                       'integration_receipt', 'local_verification_path'] and isinstance(value, str) \
                    and value.startswith(('physics/', 'formalization/')):
                result.add(value)
            if key != 'path_catalog':
                result |= collect_paths(value)
    elif isinstance(obj, list):
        for value in obj:
            result |= collect_paths(value)
    return result

def main():
    global WRITE_ATTEMPTS, WRITE_COMPLETIONS
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--plan', action='store_true', help='Print static target/guard plan without reading or changing release state.')
    parser.add_argument('--apply', action='store_true', help='Apply after actual PASS checks and reviewed authorization; otherwise preparation is read-only.')
    args = parser.parse_args()
    require(not (args.plan and args.apply), '--plan and --apply are mutually exclusive.')
    if args.plan:
        print(json.dumps(plan(), ensure_ascii=False, indent=2))
        return
    prepared, originals, protected, summary = prepare()
    if not args.apply:
        print(json.dumps(summary, ensure_ascii=False, indent=2))
        return
    for path, expected in protected.items():
        require(path.is_file() and sha(path) == expected, f'Protected input changed before apply: {path}.')
    for path, original in originals.items():
        require((path.read_bytes() if path.is_file() else None) == original, f'Metadata changed concurrently: {path}.')
    # The only repository mutation site is guarded by the exact target allowlist.
    for path in TARGETS:
        require(path.resolve().is_relative_to(B.resolve()), f'Unsafe target: {path}.')
        WRITE_ATTEMPTS += 1
        path.write_bytes(prepared[path])
        WRITE_COMPLETIONS += 1
    for path in TARGETS:
        require(path.read_bytes() == prepared[path], f'Written bytes failed immediate readback: {path}.')
    for path, expected in protected.items():
        require(sha(path) == expected, f'Protected bytes changed: {path}.')
    summary['status'] = 'MAIN_VERIFIED_METADATA_COMPLETION_PASS'
    summary['new_lean_invocations'] = 0
    summary['new_pdf_builds'] = 0
    summary['new_remote_actions'] = 0
    print(json.dumps(summary, ensure_ascii=False, indent=2))


if __name__ == '__main__':
    try:
        main()
    except (ValueError, KeyError, OSError) as exc:
        print(json.dumps({'status': 'REFUSED_OR_FAILED', 'error': str(exc),
                          'repository_write_attempts': WRITE_ATTEMPTS,
                          'completed_repository_writes': WRITE_COMPLETIONS,
                          'scope': ('Refused before repository writes.' if WRITE_ATTEMPTS == 0 else
                                    'Application failed after bounded metadata writes began; inspect the explicit target list and readback before resuming.')}, ensure_ascii=False), file=sys.stderr)
        sys.exit(2)
