def sha(data):
    return hashlib.sha256(data).hexdigest()


def record(path):
    path = Path(path)
    b = path.read_bytes()
    try:
        p = str(path.relative_to(ROOT))
    except ValueError:
        p = str(path)
    return {'path': p, 'sha256': sha(b), 'bytes': len(b)}


def load(relative):
    return json.loads((ROOT / relative).read_text())


def jsbytes(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()

def slices(text):
    marks = list(re.finditer(r'^<a id="(ch-\d+|appendix-[a-f])"></a>$',text,re.M))
    expected = ['ch-'+str(i) for i in range(60)] + ['appendix-'+s for s in 'abcdef']
    assert [m.group(1) for m in marks] == expected
    return {m.group(1):text[m.start():marks[i+1].start() if i+1<len(marks) else len(text)]
            for i,m in enumerate(marks)}


def sections(body):
    return [{'id':m.group(1),'title':m.group(2)}
            for m in re.finditer(r'^## ([0-9]+\.[0-9]+|[A-F]\.[0-9]+)\s+(.+)$',body,re.M)]


def audit_bindings(names, name_sources, generic, witness):
    by_source = collections.defaultdict(list)
    for name in dict.fromkeys(names):
        assert name in generic or name in witness, ('NOT_ACTUALLY_AUDITED',name)
        assert name in name_sources, ('NO_SOURCE_BINDING',name)
        by_source[name_sources[name]].append(name)
    out=[]
    for p,names_here in sorted(by_source.items()):
        is_witness = '/scripts/witnesses/' in p
        out.append({**record(ROOT/p),'declarations':names_here,
          'role':'DEDICATED_WITNESS' if is_witness else 'GENERIC_MODULE',
          'actual_named_audit_receipt':E+('finite-witnesses/verification.json' if is_witness else 'proof-axioms.json'),
          'all_names_present_in_actual_named_audit':True,
          'note':'Concrete witnesses are counted separately from the generic theorem/audit totals.' if is_witness else 'Explicit formal declarations; definitions, constructors and instances are not counted as physical laws.'})
    return out

    d,prior_bytes,receipt=make_dossier()
    db=jsbytes(d)
    receipt['dossier']={'path':DOSSIER,'sha256':sha(db),'bytes':len(db)}
    receipt['generator']=record(Path(__file__))
    if args.install:
        (ROOT/BE).mkdir(parents=True,exist_ok=True)
        pp=ROOT/PRIOR
        if pp.exists():assert pp.read_bytes()==prior_bytes
        else:pp.write_bytes(prior_bytes)

        # dossier mutation, to avoid accepting concurrent source/prose change.
        assert record(ROOT/(BOOK+'UEOT_PHYSICS.md'))==receipt['canonical']

            assert record(ROOT/rec['path'])['sha256']==rec['sha256']

          '# Edition3.3 dossier assembly\n\n'

These are formal declaration counts, not physical law counts. Every cited new name is matched to the real unified generic or witness audit and current source hash.\n\n'
