from pathlib import Path, PurePosixPath
import re
import zipfile


def sha256(data):
    return hashlib.sha256(data).hexdigest()


def object_sha(kind, data):
    return hashlib.sha1(kind.encode() + b' ' + str(len(data)).encode() + b'\0' + data).hexdigest()


def git_trees(entries):
    leaves = {e['path']: {'mode': e['mode'], 'type': e['type'], 'sha': e['sha']}
              for e in entries if e['type'] != 'tree'}
    directories = {''}
    for e in entries:
        if e['type'] == 'tree':
            directories.add(e['path'])
        directories.update(str(p) for p in PurePosixPath(e['path']).parents if str(p) != '.')
    trees = {}
    for directory in sorted(directories, key=lambda p: len(PurePosixPath(p).parts), reverse=True):
        children = {}
        for path, entry in leaves.items():
            parent = str(PurePosixPath(path).parent)
            if ('' if parent == '.' else parent) == directory:
                children[PurePosixPath(path).name] = entry
        for path, value in trees.items():
            parent = str(PurePosixPath(path).parent)
            if ('' if parent == '.' else parent) == directory:
                children[PurePosixPath(path).name] = {'mode': '040000', 'type': 'tree', 'sha': value}
        ordered = sorted(children, key=lambda n: (n + ('/' if children[n]['type'] == 'tree' else '')).encode())
        body = b''.join((('40000' if children[n]['type'] == 'tree' else children[n]['mode']) + ' ' + n).encode()
                        + b'\0' + bytes.fromhex(children[n]['sha']) for n in ordered)
        trees[directory] = object_sha('tree', body)
    return trees, leaves

    with zipfile.ZipFile(archive) as z:
        overlay = [n for n in z.namelist() if n.startswith('repo_overlay/') and not n.endswith('/')]
        assert len(overlay) == len(set(overlay)) == 244
        assert {n.removeprefix('repo_overlay/') for n in overlay} == {e['path'] for e in payload}
        for e in payload:
            name = e['path']
            path = PurePosixPath(name)
            assert not path.is_absolute() and '..' not in path.parts and str(path) == name
            assert name.startswith(('physics/qm/', 'formalization/ueot-qm/')) or name == '.github/workflows/ueot-qm.yml'
            assert not historical.search(name), name
            assert not any(x in path.parts for x in ('.lake','__pycache__'))
            assert path.suffix not in ('.pyc','.olean','.ilean')
            assert (root/name).is_file() and not (root/name).is_symlink()
            assert e['mode'] in ('100644', '100755')
            content = base64.b64decode(e['content'], validate=True) if e['binary'] else e['content'].encode('utf-8')
            assert e['encoding'] == ('base64' if e['binary'] else 'utf-8')
            if e['binary']:
                assert name == 'physics/qm/book/UEOT_PHYSICS.pdf'
            assert len(content) == e['bytes'] and sha256(content) == e['sha256']
            assert object_sha('blob', content) == e['git_blob_sha']
            assert content == (root/name).read_bytes() == z.read('repo_overlay/'+name), name
            previous = base_leaves.get(name)
            assert e['previous_blob'] == (previous['sha'] if previous else None)
            if previous:
                assert previous['mode'] == e['mode'] and previous['type'] == 'blob'
                assert previous['sha'] != e['git_blob_sha']
            published[name] = content
            file_rows.append({k:v for k,v in e.items() if k != 'content'})
    assert sum(e['bytes'] for e in payload) == inventory['bytes'] == 6982828
    assert sum(e['previous_blob'] is not None for e in payload) == inventory['changed_existing'] == 18
    assert sum(e['previous_blob'] is None for e in payload) == inventory['new_paths'] == 226

    book_sha = {'md':'83f9f0a470816a49db9e6c3e5d6456ac83cd191612d5691b67979cf8822c7c9c',

Root must reconcile actual current remote base, candidate tree, source PR and independent main CI. No heavy build, new theorem, physical experiment, or CI artifact ZIP inspection performed.',
