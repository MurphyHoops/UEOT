class FinalizationError(Exception):
    pass


def require(condition, message):
    if not condition:
        raise FinalizationError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def blob(data):
    return hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\0' + data).hexdigest()


def encoded(value):
    return (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode('utf-8')


class Plan:
    def __init__(self, root):
        self.root = root
        self.inputs = {}
        self.outputs = {}

    def raw(self, path):
        path = Path(path).resolve()
        data = path.read_bytes()
        previous = self.inputs.setdefault(path, data)
        require(previous == data, 'Input changed during planning: ' + str(path))
        return data

    def read(self, path):
        return json.loads(self.raw(path))

    def stage(self, path, value):
        path = Path(path).resolve()
        require(path.is_relative_to(self.root), 'Output outside requested root: ' + str(path))
        if path not in self.inputs:
            self.inputs[path] = path.read_bytes() if path.exists() else None
        self.outputs[path] = value if isinstance(value, bytes) else encoded(value)

    def verify_unchanged(self):
        for path, expected in self.inputs.items():
            actual = path.read_bytes() if path.exists() else None
            require(actual == expected, 'Input or output destination changed: ' + str(path))


def context_path(context_file, context, key, default):
    path = Path(context.get(key, default))
    return path if path.is_absolute() else context_file.parent / path

    require(c['source_count'] == 80 and c['source_set_sha256'] == SOURCE_SET, 'Wrong accepted v31.7 source identity')

            and j['verification_complete'] is True and j['repository'] == REPOSITORY
            and j['event'] == kind and j['run_id'] == c[kind + '_run'] and j['source_commit'] == commit,
            kind + ': wrong or incomplete actual CI receipt')
        require(j['generic_axiom_reports'] == 855 and j['runtime_test_methods'] == 100
            and j['finite_witness_total_audits'] == 151 and j['printed_source_hash_count'] == 33,
            kind + ': incorrect accepted gate counts')

            and j['local_accepted_source_count'] == 80 and j['local_accepted_source_set_sha256'] == SOURCE_SET,
            kind + ': incorrect witness or source identity')

            'generic_axiom_names_and_foundation_equal_local', 'archive_steps_success',
            'input_files_unchanged_during_verification', 'local_accepted_sources_unchanged_during_verification']:

            and all(j['required_steps_success'].values()), kind + ': missing successful required steps')

            data = plan.raw(Path(bound['path']))
            require(sha(data) == bound['sha256'] and len(data) == bound['bytes'],
                kind + ': supplied CI or reference bytes changed: ' + key)
        actual_run = plan.read(evidence / 'ci' / f'{kind}-{c[kind + "_run"]}-run.json')
        require(actual_run['id'] == c[kind + '_run'] and actual_run['head_sha'] == commit
            and actual_run['status'] == 'completed' and actual_run['conclusion'] == 'success'
            and actual_run['event'] == ('pull_request' if kind == 'pr' else 'push'),

        require(kind != 'main' or actual_run['head_branch'] == 'main', 'Main receipt is not a main push')

    tree_files = {

            'source_commit': c['source_commit'], 'source_tree': c['source_tree'], 'source_count': 80,

            'scope': 'Actual recursive source-tree readback reconciled with all 244 frozen published paths, 80 accepted inputs, 56 imported Core sources and 3 canonical artifacts. Recorded after the source and main checks; source_and_main_match fields include that separate reconciliation.'})

        'schema': 'ueot_qm.remote_main_reconciliation.v31.7', 'status': 'PASS', 'checked_at_utc': now,
        'source_commit': c['source_commit'], 'source_tree': c['source_tree'],
        'tested_main_commit': c['tested_main_commit'], 'tested_main_tree': c['tested_main_tree'],

        'published_parent_commit': c['parent_commit'], 'published_parent_tree': c['parent_tree'],

        'pull_request': c['pull_request'], 'pull_request_url': c['pull_request_url'],
        'pr_run': c['pr_run'], 'pr_run_url': c['pr_run_url'], 'pr_status': 'PASS', 'pr_verification': prpath,

        'main_run': c['main_run'], 'main_run_url': c['main_run_url'], 'main_status': 'PASS', 'main_verification': mainpath,
        'source_and_main_tree_identical': tree_equal, 'tested_inputs_identical': True,
        'core_imports_identical': True, 'canonical_book_identical': True, 'published_paths_identical': True,
        'published_244_paths_identical': True, 'published_path_count': count, 'source_count': 80, 'source_set_sha256': SOURCE_SET,
        'source_manifest': 'physics/qm/kernel/evidence/v31.7/source-manifest.json',
        'source_reconciliation': 'physics/qm/kernel/evidence/v31.7/remote-source-reconciliation.json',
        'main_reconciliation': 'physics/qm/kernel/evidence/v31.7/remote-main-reconciliation.json',
        'local_machine_acceptance': 'PASS', 'remote_pr': 'PASS', 'resulting_main': 'PASS',
        'generic_modules': 29, 'explicit_theorems': 646, 'named_axiom_reports': 855,
        'finite_witness_audit_reports': 151, 'finite_witness_theorems': 110,
        'finite_witness_definitions': 39, 'finite_witness_abbreviations': 2,

        'pr_build_jobs': receipts['pr']['build_jobs'], 'main_build_jobs': receipts['main']['build_jobs'],

        'scope': ['physics/qm/**', 'formalization/ueot-qm/**', '.github/workflows/ueot-qm.yml'],

    current['integration'].update(state='MAIN_VERIFIED', temporary_branch=c['source_branch'],
        source_commit=c['source_commit'], source_tree=c['source_tree'], pull_request=c['pull_request'],
        pull_request_url=c['pull_request_url'], pr_run=c['pr_run'], pr_run_url=c['pr_run_url'],

The later evidence/navigation metadata commit is not another scientific execution or the tested source commit.',

    old_claims = copy.deepcopy(ledger['claims'])
    ledger['metadata_updated_at_utc'] = now

Current 56 imported Core source bytes match the frozen v31.7 acceptance and actual source {c["source_commit"]}/tested main {c["tested_main_commit"]}. The published research baseline is {BASELINE}. No frozen Core mutation by this increment.'

    active['historical_checkpoint'] = {'scope': 'Exact prior checkpoint preserved as historical development progress, not the current acceptance state.',
        'record_updated_at_utc': active.get('updated_at_utc'), 'checkpoint': copy.deepcopy(active['checkpoint'])}
    active['checkpoint'] = {
        'new_main_sources': 'All seven v31.7 modules are in the source-bound accepted package: 29 modules, 646 explicit theorems, 855 named audits.',

        'book': 'Formal monograph 3.3 local release is complete: 203 pages, 60 chapters, 6 appendices, 291 sections and 131 results. Its unchanged three canonical artifacts match the actual source and tested main; book remote metadata is recorded separately.',
        'execution': f'Local acceptance remains its original receipt. Actual PR run {c["pr_run"]} and resulting-main run {c["main_run"]} passed; metadata finalization runs no scientific gate.',
        'next_plan': 'UEOT_QM_EXECUTION_v31.8.md remains unformalized planning, not part of the v31.7 proof counts.'}

        q / 'UEOT_QM_KERNEL_v31.7.md', q / 'UEOT_QM_EXECUTION_v31.7.md', root / 'formalization/ueot-qm/README.md']
    for path in documentation:
        text = plan.raw(path).decode('utf-8')
        require(text.count('LOCAL_VERIFIED_REMOTE_PENDING') == 1, 'Expected one current status marker: ' + str(path))
        text = text.replace('LOCAL_VERIFIED_REMOTE_PENDING', 'MAIN_VERIFIED', 1)

    protected = {str((root / s['path']).resolve()) for s in manifest['sources'] + core['modules']}
    protected.update(str((q / 'book' / name).resolve()) for name in CANONICAL)
    require(not protected.intersection(str(p) for p in plan.outputs), 'Proposed output would change scientific or canonical bytes')
    plan.verify_unchanged()
    summary = {'status': 'READY_TO_APPLY', 'source_commit': c['source_commit'], 'tested_main_commit': c['tested_main_commit'],
        'source_count': 80, 'core_imports': 56, 'publication_paths': count, 'canonical_artifacts': 3,
        'tree_equal': tree_equal, 'no_scientific_execution': True,
        'outputs': [{'path': p.relative_to(root).as_posix(), 'bytes': len(data), 'sha256': sha(data)} for p, data in plan.outputs.items()]}
    return plan, summary


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--root', type=Path, default=Path('/workspace/scratch/f11b4261e187/qm_v317_work'))
    parser.add_argument('--context', type=Path, default=Path(__file__).resolve().parent / 'qm317_release_context.json')
    modes = parser.add_mutually_exclusive_group()
    modes.add_argument('--apply', action='store_true')
    modes.add_argument('--dry-run', action='store_true', help='Explicit read-only mode (the default)')
    args = parser.parse_args()
    try:
        plan, summary = build_plan(args.root.resolve(), args.context.resolve())
        if args.apply:
            # All bytes and destinations were validated above; recheck immediately before writes.
            plan.verify_unchanged()
            temporary = []
            try:
                for path, data in plan.outputs.items():
                    path.parent.mkdir(parents=True, exist_ok=True)
                    with tempfile.NamedTemporaryFile(dir=path.parent, prefix='.' + path.name + '.', delete=False) as f:
                        f.write(data)
                        staged = Path(f.name)
                    staged.chmod(path.stat().st_mode & 0o777 if path.exists() else 0o644)
                    temporary.append((staged, path))
                plan.verify_unchanged()
                for staged, path in temporary:
                    staged.replace(path)
                summary['status'] = 'MAIN_VERIFIED_METADATA_APPLIED'
            finally:
                for staged, _ in temporary:
                    staged.unlink(missing_ok=True)
        print(json.dumps(summary, ensure_ascii=False))
        return 0
    except (FinalizationError, OSError, ValueError, KeyError, TypeError, StopIteration) as error:
        print(json.dumps({'status': 'BLOCKED', 'error': type(error).__name__ + ': ' + str(error),
            'scientific_execution_performed': False}, ensure_ascii=False))
        return 1


if __name__ == '__main__':
    sys.exit(main())
