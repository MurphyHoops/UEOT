\[
\int g^2(N_D-N_{D,0})\,d\tau
\ge-\frac1\pi\int_0^\infty
\widehat{\mathcal F_g}(\beta,-\beta)\,d\beta ,

\[
\mathcal F_g(\tau,\tau')=
g(\tau)g(\tau')(\ell\cdot\nabla)(\ell'\cdot\nabla')
W_{D,0}(\gamma(\tau),\gamma(\tau')).
